import { useState, useEffect } from 'react'; import FlightList from './components/FlightList'; import AddFlightForm from './components/AddFlightForm'; import FilterBar from './components/FilterBar'; function App() { const [flights, setFlights] = useState<Flight[]>([]); const [filter, setFilter] = useState<FlightStatus | 'all'>('all'); // טעינה מ-localStorage useEffect(() => { const saved = localStorage.getItem('flights'); if (saved) setFlights(JSON.parse(saved)); }, []); // שמירה ב-localStorage useEffect(() => { localStorage.setItem('flights', JSON.stringify(flights)); }, [flights]); const addFlight = (flight: Flight) => { setFlights(prev => [...prev, flight]); }; const deleteFlight = (id: string) => { setFlights(prev => prev.filter(f => f.id !== id)); }; const updateStatus = (id: string, status: FlightStatus) => { setFlights(prev => prev.map(f => f.id === id ? { ...f, status } : f )); }; const filteredFlights = filter === 'all' ? flights : flights.filter(f => f.status === filter); return ( <div className="app"> <h1>מערכת בקרת טיסות</h1> <Dashboard flights={flights} /> <FilterBar filter={filter} onChange={setFilter} /> <AddFlightForm onAdd={addFlight} /> <FlightList flights={filteredFlights} onDelete={deleteFlight} onStatusChange={updateStatus} /> </div> ); }

שלב 4: שאלת JS — ספירת מספרים במטריצה

השאלה

בהינתן מטריצה של מספרים (מערך דו-ממדי), כתוב פונקציה שמחזירה כמה מספרים יש בכל שורה ובכל עמודה שגדולים מ-50.

function countBig(matrix: number[][]) { const rows = matrix.length; const cols = matrix[0]?.length || 0; const rowCounts: number[] = []; const colCounts: number[] = new Array(cols).fill(0); for (let i = 0; i < rows; i++) { let rowCount = 0; for (let j = 0; j < cols; j++) { if (matrix[i][j] > 50) { rowCount++; colCounts[j]++; } } rowCounts.push(rowCount); } return { rowCounts, colCounts }; } // בדיקה const m = [ [10, 60, 30], [70, 80, 20], [40, 90, 100] ]; countBig(m); // { // rowCounts: [1, 2, 2], // כמה גדולים בכל שורה // colCounts: [1, 3, 1] // כמה גדולים בכל עמודה // }
טיפ: תמיד התחל עם מקרי קצה — מטריצה ריקה, שורה ריקה, ערכים שליליים. השתמש ב-matrix[0]?.length || 0 להגנה.

שלב 1: Classes

class Account { id: string; type: 'checking' | 'savings'; balance: number; owner: string; constructor(id: string, type: 'checking' | 'savings', owner: string, balance = 0) { this.id = id; this.type = type; this.owner = owner; this.balance = balance; } deposit(amount: number): void { if (amount <= 0) { throw new Error('סכום ההפקדה חייב להיות חיובי'); } this.balance += amount; } withdraw(amount: number): void { if (amount <= 0) { throw new Error('סכום המשיכה חייב להיות חיובי'); } if (amount > this.balance) { throw new Error('אין מספיק יתרה'); } this.balance -= amount; } transfer(target: Account, amount: number): void { this.withdraw(amount); target.deposit(amount); } } class Customer { id: string; name: string; accounts: Account[]; constructor(id: string, name: string) { this.id = id; this.name = name; this.accounts = []; } addAccount(account: Account): void { this.accounts.push(account); } totalBalance(): number { return this.accounts.reduce((sum, acc) => sum + acc.balance, 0); } findAccount(id: string): Account | undefined { return this.accounts.find(acc => acc.id === id); } }

השאלה

בהינתן מספר (למשל 4231), החזר את אותו מספר עם ספרות ממוינות מהקטנה לגדולה (1234).

function sortDigits(num: number): number { // 1. הפוך למחרוזת ואז למערך של תווים const digits = num.toString().split(''); // 2. מיין כמספרים (לא כtext!) digits.sort((a, b) => Number(a) - Number(b)); // 3. הצמד חזרה והחזר כמספר return Number(digits.join('')); } // בדיקות sortDigits(4231); // 1234 sortDigits(9876); // 6789 sortDigits(1000); // 0001 → 1 ⚠️ sortDigits(305); // 035 → 35 ⚠️

שלב 3: שאלת JS — פלינדרום בינארי

השאלה

קלט: מספר עשרוני. פלט: true אם הייצוג הבינארי שלו פלינדרום, אחרת false.

לדוגמה: 9 = 1001 = פלינדרום ✅. 10 = 1010 = לא פלינדרום ❌.

function isBinaryPalindrome(num: number): boolean { // 1. המר לבינארי const binary = num.toString(2); // 2. בדוק פלינדרום for (let i = 0; i < Math.floor(binary.length / 2); i++) { if (binary[i] !== binary[binary.length - 1 - i]) { return false; } } return true; } // או דרך פשוטה יותר function isBinaryPalindrome(num: number): boolean { const binary = num.toString(2); const reversed = binary.split('').reverse().join(''); return binary === reversed; } // בדיקות isBinaryPalindrome(9); // true (1001) isBinaryPalindrome(10); // false (1010) isBinaryPalindrome(7); // true (111) isBinaryPalindrome(5); // true (101) isBinaryPalindrome(0); // true (0) isBinaryPalindrome(1); // true (1)
טיפ: .toString(2) ממיר לבינארי. .toString(16) = הקס. .toString(8) = אוקטל.

// rates.ts type Currency = 'USD' | 'EUR' | 'GBP' | 'ILS'; const exchangeRates: Record<Currency, Record<Currency, number>> = { USD: { USD: 1, EUR: 0.92, GBP: 0.79, ILS: 3.65 }, EUR: { USD: 1.09, EUR: 1, GBP: 0.86, ILS: 3.97 }, GBP: { USD: 1.27, EUR: 1.16, GBP: 1, ILS: 4.62 }, ILS: { USD: 0.27, EUR: 0.25, GBP: 0.22, ILS: 1 }, }; function convert(amount: number, from: Currency, to: Currency): number { if (!exchangeRates[from] || !exchangeRates[from][to]) { throw new Error('מטבע לא נתמך'); } return amount * exchangeRates[from][to]; }

שלב 4: שאלת JS — מיקום בפיבונאצ'י

השאלה

בהינתן מספר n, החזר את האיבר ה-n בסדרת פיבונאצ'י. הסדרה: 0, 1, 1, 2, 3, 5, 8, 13, 21, ...

// פתרון איטרטיבי (O(n)) — מומלץ function fibonacci(n: number): number { if (n < 0) throw new Error('n חייב להיות חיובי'); if (n <= 1) return n; let prev = 0, curr = 1; for (let i = 2; i <= n; i++) { [prev, curr] = [curr, prev + curr]; } return curr; } // בדיקות fibonacci(0); // 0 fibonacci(1); // 1 fibonacci(2); // 1 fibonacci(5); // 5 fibonacci(10); // 55 fibonacci(20); // 6765 // פתרון רקורסיבי (איטי, O(2^n)) — להראות שמכירים function fibRecursive(n: number): number { if (n <= 1) return n; return fibRecursive(n - 1) + fibRecursive(n - 2); } // פתרון רקורסיבי עם memoization (מהיר) function fibMemo(n: number, cache: Record<number, number> = {}): number { if (n <= 1) return n; if (cache[n] !== undefined) return cache[n]; cache[n] = fibMemo(n - 1, cache) + fibMemo(n - 2, cache); return cache[n]; }

שאלת בונוס: מיקום מספר בסדרה

// בהינתן מספר, מצא באיזה מיקום הוא בסדרה. אם לא קיים — החזר -1 function fibIndex(num: number): number { if (num < 0) return -1; if (num === 0) return 0; if (num === 1) return 1; let prev = 0, curr = 1, index = 1; while (curr <= num) { if (curr === num) return index; [prev, curr] = [curr, prev + curr]; index++; } return -1; // לא בסדרה } // בדיקות fibIndex(8); // 6 (0,1,1,2,3,5,8) fibIndex(13); // 7 fibIndex(4); // -1 (לא בסדרה) fibIndex(55); // 10
טיפ למבחן: תמיד תכתוב גם פתרון איטרטיבי וגם רקורסיבי. הפתרון הרקורסיבי "יפה" אבל איטי. הציון הולך לפתרון יעיל.
