⚛️ React Memoization — מניעת רינדורים מיותרים

React מרנדר (מצייר מחדש) את הקומפוננטה בכל פעם שה-state שלה משתנה. לפעמים קומפוננטה מרונדרת בלי צורך — וזה איטי. Memoization = לזכור תוצאה ולא לחשב שוב אם הקלט לא השתנה.

שלוש כלים עיקריים: React.memo (מונע רינדור של קומפוננטה שלמה), useMemo (מונע חישוב יקר), useCallback (מונע יצירת פונקציה חדשה בכל רינדור).

React.memo — עוטף קומפוננטה

// בלי memo — מרנדר בכל פעם שה-Parent מרנדר const ChildComponent = ({ name }) => { console.log('Child rendered!'); return
שלום {name}
; }; // עם memo — מרנדר רק אם ה-prop השתנה const ChildComponent = React.memo(({ name }) => { console.log('Child rendered!'); return
שלום {name}
; }); // Parent function Parent() { const [count, setCount] = useState(0); return ( <> <button onClick={() => setCount(c => c + 1)}>Count: {count}</button> <ChildComponent name="תל" /> {/* לא מרנדר שוב! */} </> ); }

useMemo — זוכר תוצאת חישוב

function ProductList({ products, filter }) { // בלי useMemo — מחושב מחדש בכל רינדור (יקר!) // const filtered = products.filter(p => p.category === filter); // עם useMemo — מחושב מחדש רק כש-products או filter משתנה const filtered = useMemo(() => { console.log('Filtering...'); return products.filter(p => p.category === filter); }, [products, filter]); // ← dependencies return ( <ul> {filtered.map(p => <li key={p.id}>{p.name}</li>)} </ul> ); }

useCallback — זוכר פונקציה

function Parent() { const [count, setCount] = useState(0); // בלי useCallback — פונקציה חדשה נוצרת בכל רינדור // const handleClick = () => console.log('clicked'); // עם useCallback — אותה פונקציה, לא גורם ל-Child לרנדר מחדש const handleClick = useCallback(() => { console.log('clicked'); }, []); // [] = לא משתנה לעולם return <MemoChild onClick={handleClick} />; } const MemoChild = React.memo(({ onClick }) => { console.log('Child rendered'); return <button onClick={onClick}>לחץ</button>; });

מה ההבדל בין useMemo ל-useCallback? (useMemo = מזכיר ערך מוחזר. useCallback = מזכיר את הפונקציה עצמה).

🧠 Memory Leaks & Garbage Collection

כל משתנה ש-JavaScript יוצר תופס מקום בזיכרון. כשאין לו יותר שימוש, ה-GC (Garbage Collector) מנקה אותו. Memory Leak = הזיכרון לא מנוקה למרות שאין לו שימוש — כי יש Reference חבוי שמחזיק אותו.

הסיבות הנפוצות ל-Memory Leak

// ❌ 1. Event Listener שלא מוסר function attachListener() { const bigData = new Array(1000000).fill('🔴'); document.addEventListener('click', () => { console.log(bigData.length); // bigData לא יגוּדל כל עוד ה-listener קיים! }); } // ✅ תיקון — לשמור reference ולהסיר function attachListener() { const bigData = new Array(1000000).fill('🟢'); const handler = () => console.log(bigData.length); document.addEventListener('click', handler); // לנקות כשלא צריך return () => document.removeEventListener('click', handler); } // ❌ 2. setInterval בלי clearInterval function startTimer() { const data = fetchHugeData(); setInterval(() => process(data), 1000); // data לא יגוּדל לעולם! } // ✅ תיקון function startTimer() { const data = fetchHugeData(); const id = setInterval(() => process(data), 1000); return () => clearInterval(id); // cleanup }

Memory Leak ב-React (useEffect)

// ❌ בעיה — setState אחרי שה-component unmounted function UserProfile({ userId }) { const [user, setUser] = useState(null); useEffect(() => { fetch(`/api/users/${userId}`) .then(r => r.json()) .then(data => setUser(data)); // 💥 אם נסגר לפני שה-fetch הסתיים! }, [userId]); } // ✅ תיקון — AbortController function UserProfile({ userId }) { const [user, setUser] = useState(null); useEffect(() => { const controller = new AbortController(); fetch(`/api/users/${userId}`, { signal: controller.signal }) .then(r => r.json()) .then(data => setUser(data)) .catch(err => { if (err.name !== 'AbortError') console.error(err); }); return () => controller.abort(); // cleanup! }, [userId]); }

WeakMap / WeakRef — References חלשים

// WeakMap — לא מונע GC מלנקות את ה-key! const cache = new WeakMap(); function processElement(el) { if (cache.has(el)) return cache.get(el); const result = heavyComputation(el); cache.set(el, result); return result; // כשה-element נמחק מה-DOM, ה-entry ב-WeakMap מנוקה אוטומטית! } // Map רגילה — תחזיק את ה-element לנצח (leak) const badCache = new Map(); badCache.set(el, result); // el לא יגוּדל גם אחרי שנמחק מה-DOM
למה Closure עלול לגרום ל-Memory Leak? (כי פונקציית Closure מחזיקה reference לסביבה החיצונית שלה — כולל משתנים גדולים — גם כשלא צריך אותם יותר).

🌊 Node.js Streams & Buffers

דמיין שאתה מוזג מים מבקבוק לכוס. אפשר לשפוך הכל בבת אחת (= fs.readFile) — אבל אם הבקבוק ענק, הכוס תשפך. Stream = שופכים טיפה-טיפה, בלי לטעון הכל לזיכרון.

Streams חיוניים לכל פעולת קלט/פלט גדולה: קריאת קבצים, upload/download, WebSockets, תשובות HTTP. יש 4 סוגי Streams: Readable, Writable, Duplex, Transform.

fs.readFile vs fs.createReadStream

const fs = require('fs'); // ❌ readFile — טוען את כל הקובץ לRAM! // לקובץ של 1GB — צריך 1GB+ זיכרון fs.readFile('huge.csv', 'utf-8', (err, data) => { console.log(data); // 1GB בזיכרון! }); // ✅ createReadStream — מעבד chunk-by-chunk // זיכרון קבוע גם לקובץ של 100GB! const stream = fs.createReadStream('huge.csv', { encoding: 'utf-8', highWaterMark: 64 * 1024 // 64KB chunks }); stream.on('data', (chunk) => { process(chunk); // מעבד כל חתיכה }); stream.on('end', () => console.log('Done!')); stream.on('error', (err) => console.error(err));

pipe — ניקוז Stream לStream

const fs = require('fs'); const zlib = require('zlib'); // Stream Pipeline: קרא קובץ → דחוס → שמור // הכל ב-64KB בכל פעם, לא משנה גודל הקובץ! fs.createReadStream('input.txt') .pipe(zlib.createGzip()) // Transform Stream — דחיסה .pipe(fs.createWriteStream('output.txt.gz')) .on('finish', () => console.log('Compressed!')); // ✅ stream.pipeline (Node 10+) — יותר בטוח מ-pipe const { pipeline } = require('stream/promises'); await pipeline( fs.createReadStream('input.txt'), zlib.createGzip(), fs.createWriteStream('output.txt.gz') );

Streams ב-Express — response Streaming

// במקום לטעון קובץ PDF ענק לזיכרון ולשלוח: app.get('/download', (req, res) => { const filePath = './reports/huge-report.pdf'; // ✅ Stream ישירות לתשובה — חוסך RAM בשרת res.setHeader('Content-Type', 'application/pdf'); fs.createReadStream(filePath).pipe(res); }); // Buffer — מחזיק בינאריים בזיכרון const buf = Buffer.from('Hello', 'utf-8'); console.log(buf); // <Buffer 48 65 6c 6c 6f> console.log(buf.toString('utf-8')); // 'Hello' console.log(buf.toString('base64')); // 'SGVsbG8='
גישהזיכרוןמתאים ל
fs.readFileגודל הקובץ כולוקבצים קטנים (<10MB)
createReadStreamchunk בלבד (~64KB)קבצים גדולים, CSV, JSON ענקי
pipe/pipelinechunk בלבדשרשור פעולות (קרא→דחוס→שמור)
מה ההבדל בין Buffer ל-Stream? (Buffer = נתונים בינאריים בזיכרון. Stream = ממשק לקריאה/כתיבה של נתונים ב-chunks לאורך זמן).

🐳 Docker Containerization

Docker = קופסה קסומה שאומרת "אצלי זה עובד ➔ אצלכם זה גם יעבוד". הקופסה מכילה את הקוד, Node.js, התלויות — הכל. מריצים אותה בכל מחשב/שרת בלי הפתעות.

Dockerfile — מתכון לקופסה

# ── server/Dockerfile ────────────────────────────── # 1. Image בסיס (Operating System + Node) FROM node:20-alpine # 2. תיקיית עבודה בתוך ה-container WORKDIR /app # 3. העתק package.json ותתקן תלויות COPY package*.json ./ RUN npm install --production # 4. העתק את שאר הקוד COPY . . # 5. חשוף את ה-port EXPOSE 5000 # 6. הרץ את השרת CMD ["node", "src/server.js"]

▲ Next.js — SSR & SSG

React בנה SPA — הדף ריק עד שJS נטען. Next.js מוסיף: רינדור בשרת (SSR) ודפים סטטיים (SSG) — מהיר יותר, SEO טוב יותר.

SSR vs SSG vs CSR

שיטהמתי HTML נבנה?מתאים ל
CSR (React)בdפדפן, אחרי JSdashboards, apps פרטיות
SSG (Static)build time — פעם אחתבלוג, docs, landing page
SSR (Server)בכל request בשרתחנות, feed ב-real-time

getServerSideProps — SSR

// pages/products/[id].js export async function getServerSideProps(context) { const { id } = context.params; const res = await fetch(`https://api.example.com/products/${id}`); const product = await res.json(); return { props: { product }, // מועבר לcomponent }; } export default function ProductPage({ product }) { return <h1>{product.name}</h1>; }

getStaticProps + getStaticPaths — SSG

// pages/blog/[slug].js // מגדיר אילו paths לבנות ב-build time export async function getStaticPaths() { const posts = await fetchAllPosts(); return { paths: posts.map(p => ({ params: { slug: p.slug } })), fallback: false, // 404 לpaths לא-מוגדרים }; } // מביא data לכל path export async function getStaticProps({ params }) { const post = await fetchPost(params.slug); return { props: { post } }; } export default function BlogPost({ post }) { return <article>{post.content}</article>; }

App Router (Next.js 13+) — Server Components

// app/users/page.js — Server Component כברירת מחדל // אין 'use client' — רץ בשרת בלבד, אפס JS לclient async function UsersPage() { // גישה ישירה לDB — בלי API route! const users = await prisma.user.findMany(); return ( <ul> {users.map(u => <li key={u.id}>{u.name}</li>)} </ul> ); } // app/counter/page.js — Client Component 'use client'; // חובה כשמשתמשים ב-hooks/events import { useState } from 'react'; export default function Counter() { const [count, setCount] = useState(0); return <button onClick={() => setCount(c => c+1)}>{count}</button>; }
מה ההבדל בין SSR לSSG? SSG נבנה פעם אחת ב-build — מהיר מאוד. SSR נבנה בכל request — תמיד מעודכן אבל איטי יותר.

🗃️ Prisma ORM

Prisma = מתרגם בין TypeScript לDB. במקום SQL גולמי, כותבים TypeScript ו-Prisma בונה את ה-SQL. Type-safe, auto-complete, ו-migrations אוטומטיות.

schema.prisma — הגדרת מודל

// prisma/schema.prisma datasource db { provider = "postgresql" // או mongodb, mysql, sqlite url = env("DATABASE_URL") } generator client { provider = "prisma-client-js" } model User { id Int @id @default(autoincrement()) email String @unique name String? posts Post[] // relation createdAt DateTime @default(now()) } model Post { id Int @id @default(autoincrement()) title String author User @relation(fields: [authorId], references: [id]) authorId Int }

Prisma Client — CRUD

import { PrismaClient } from '@prisma/client'; const prisma = new PrismaClient(); // Create const user = await prisma.user.create({ data: { email: 'tal@example.com', name: 'Tal' } }); // Read — עם filter const adults = await prisma.user.findMany({ where: { age: { gte: 18 } }, orderBy: { name: 'asc' }, take: 10, // pagination }); // Read עם relation — כמו JOIN const userWithPosts = await prisma.user.findUnique({ where: { id: 1 }, include: { posts: true } // כמו populate() ב-Mongoose }); // Update await prisma.user.update({ where: { id: 1 }, data: { name: 'Tal Updated' } }); // Delete await prisma.user.delete({ where: { id: 1 } });

Prisma Migrate — סנכרון Schema עם DB

# יוצר migration file + מעדכן DB npx prisma migrate dev --name add_user_table # הפעל Prisma Studio (UI ויזואלי לDB) npx prisma studio
Prisma vs Mongoose: Prisma מיועד ל-SQL (PostgreSQL, MySQL) עם type safety מלא. Mongoose מיועד ל-MongoDB. שניהם ORM/ODM אבל לסוגי DB שונים.
מה יתרון Prisma על SQL גולמי? Type safety — TypeScript יידע בזמן compile אם query שגוי. Auto-complete. Migrations אוטומטיות.
