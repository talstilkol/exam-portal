type FlightStatus = 'scheduled' | 'boarding' | 'departed' | 'arrived' | 'delayed';

type Flight = {
  id: string;
  number: string;
  from: string;
  to: string;
  departureTime: string;
  status: FlightStatus;
  passengers: number;
  gate: string;
};

function App() {
  const [flights, setFlights] = useState<Flight[]>([]);
  const [filter, setFilter] = useState<FlightStatus | 'all'>('all');

  useEffect(() => {
    const saved = localStorage.getItem('flights');
    if (saved) setFlights(JSON.parse(saved));
  }, []);

  useEffect(() => {
    localStorage.setItem('flights', JSON.stringify(flights));
  }, [flights]);

  const addFlight = (flight: Flight) => {
    setFlights(prev => [...prev, flight]);
  };

  const updateStatus = (id: string, status: FlightStatus) => {
    setFlights(prev => prev.map(f =>
      f.id === id ? { ...f, status } : f
    ));
  };

  const filteredFlights = filter === 'all'
    ? flights
    : flights.filter(f => f.status === filter);

  return <FlightList flights={filteredFlights} onStatusChange={updateStatus} />;
}

שאלת JS — מטריצה

countBig — פתרון שאלת האלגוריתם
function countBig(matrix: number[][], threshold = 50) {
  const rows = matrix.length;
  const cols = matrix[0]?.length || 0;
  const rowCounts = new Array(rows).fill(0);
  const colCounts = new Array(cols).fill(0);

  for (let i = 0; i < rows; i++) {
    for (let j = 0; j < cols; j++) {
      if (matrix[i][j] > threshold) {
        rowCounts[i]++;
        colCounts[j]++;
      }
    }
  }

  return { rowCounts, colCounts };
}
בדיקת חובה
בדוק מטריצה 3x3, ערך סף 50, והשווה ידנית את rowCounts ו־colCounts.
נפילה נפוצה
לשכוח לאתחל colCounts לפי מספר העמודות, ואז colCounts[j] יהיה undefined.
מה לכתוב בהסבר
הסיבוכיות היא O(rows × cols) כי עוברים על כל תא בדיוק פעם אחת.

class Account {
  constructor(id, type, balance = 0) {
    this.id = id;
    this.type = type;
    this.balance = balance;
  }

  deposit(amount) {
    if (amount <= 0) throw new Error('Invalid amount');
    this.balance += amount;
    return this.balance;
  }

  withdraw(amount) {
    if (amount <= 0) throw new Error('Invalid amount');
    if (amount > this.balance) throw new Error('Insufficient funds');
    this.balance -= amount;
    return this.balance;
  }
}

class Customer {
  constructor(id, name, accounts = []) {
    this.id = id;
    this.name = name;
    this.accounts = accounts;
  }

  totalBalance() {
    return this.accounts.reduce((sum, account) => sum + account.balance, 0);
  }
}

function sortDigits(num: number): number {
  return Number(
    Math.abs(num)
      .toString()
      .split('')
      .sort((a, b) => Number(a) - Number(b))
      .join('')
  );
}

class Inventory {
  constructor() {
    this.products = [];
  }

  addProduct(product) {
    const exists = this.products.some(p => p.id === product.id);
    if (exists) throw new Error('Product already exists');
    this.products.push(product);
  }

  searchByName(text) {
    return this.products.filter(p =>
      p.name.toLowerCase().includes(text.toLowerCase())
    );
  }
}

fibonacciIndex
function fibonacciIndex(target: number): number {
  if (target === 0) return 0;
  if (target === 1) return 1;

  let prev = 0;
  let current = 1;
  let index = 1;

  while (current < target) {
    const next = prev + current;
    prev = current;
    current = next;
    index++;
  }

  return current === target ? index : -1;
}

function countBig(matrix: number[][], threshold = 50) {
  const rows = matrix.length;
  const cols = matrix[0]?.length || 0;
  const rowCounts = new Array(rows).fill(0);
  const colCounts = new Array(cols).fill(0);

  for (let i = 0; i < rows; i++) {
    for (let j = 0; j < cols; j++) {

        rowCounts[i]++;
        colCounts[j]++;
      }
    }
  }

  return { rowCounts, colCounts };
}

function fibonacciIndex(target: number): number {
  if (target === 0) return 0;
  if (target === 1) return 1;

  let prev = 0;
  let current = 1;
  let index = 1;

  while (current < target) {
    const next = prev + current;
    prev = current;
    current = next;
    index++;
  }

  return current === target ? index : -1;
}

function countSequences(arr: number[], target: number) {
  let count = 0;

  for (let i = 0; i < arr.length; i++) {
    let sum = 0;
    for (let j = i; j < arr.length; j++) {
      sum += arr[j];
      if (j > i && sum === target) count++;
    }
  }

  return count;
}

  const seen = new Set<number>();
  const result: number[] = [];

    if (!seen.has(item)) {
      seen.add(item);
      result.push(item);
    }
  }

  return result;
