2// "סכום לפי אורך מספר" — function takes an array of numbers and returns
3// an object whose keys are digit-lengths and values are the sums of numbers
4// with that digit-length.
5//
6// Example:
7// sumByDigitLength([1, 15, 30, 5, 800]) → { 1: 6, 2: 45, 3: 800 }
8// (1+5=6 are 1-digit; 15+30=45 are 2-digit; 800 is 3-digit)
9//
10// Spec note: throws a relevant error if any value is not a number.
11 
12/**
13 * @param {number[]} arr
14 * @returns {Record<number, number>}
15 * @throws {TypeError} if any element is not a finite number
16 */
17function sumByDigitLength(arr) {
18 if (!Array.isArray(arr)) {
19 throw new TypeError(`expected an array, got ${typeof arr}`);
20 }
21 const out = {};
22 for (let i = 0; i < arr.length; i++) {
23 const v = arr[i];
24 // ✅ FIXED: throw a relevant error per spec for non-numbers
25 if (typeof v !== "number" || !Number.isFinite(v)) {
26 throw new TypeError(
27 `value at index ${i} is not a finite number: ${JSON.stringify(v)}`
28 );
29 }
30 // ✅ FIXED: use Math.abs so negative numbers don't count the "-" sign
31 // and so -0 and 0 are equivalent
32 const len = String(Math.abs(v)).length;
33 out[len] = (out[len] || 0) + v;
34 }
35 return out;
36}
