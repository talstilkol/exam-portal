45function loadCurrencies() {
46 try {

49 const arr = JSON.parse(raw);

271function UpdateCurrency({ currencies, setCurrencies }) {
272 const [type, setType] = useState("");
273 const [value, setValue] = useState("");
274 const [errors, setErrors] = useState({});
275 const navigate = useNavigate();
276 
277 // ✅ UPDATE button disabled until both fields filled
278 const canUpdate =
279 type !== "" && value !== "" && !Number.isNaN(Number(value));
280 
281 function setField(field, val) {
282 if (field === "type") setType(val);
283 else setValue(val);
284 if (errors[field]) {
285 setErrors((e) => {
286 const c = { ...e };
287 delete c[field];
288 return c;
289 });
290 }
291 }
292 
293 function update() {
294 const errs = {};
295 // ✅ FIXED: English letters only validation per spec
296 if (!type || !ENGLISH_ONLY.test(type))
297 errs.type = "Currency type must be English letters only";
298 const num = Number(value);
299 if (value === "" || Number.isNaN(num))
300 errs.value = "Value must be a number";
301 
302 if (Object.keys(errs).length > 0) {
303 setErrors(errs);
304 return;
305 }
306 
307 // ✅ FIXED: if currency type already exists → call .update() (NOT push duplicate)
308 const upper = type.toUpperCase();
309 const existing = currencies.find((c) => c.type === upper);
310 let next;
311 if (existing) {
312 // mutate the instance and return a new array so React re-renders
313 existing.update(num);
314 next = [...currencies];
315 } else {
316 // create new instance
317 next = [...currencies, new Currency(upper, num)];
318 }
319 setCurrencies(next);
320 alert(
321 existing
322 ? `Updated ${upper} value to ${num}`
323 : `Created new currency ${upper} = ${num}`
324 );
325 setType("");
326 setValue("");
327 setErrors({});
328 }
329

332 <header className="hero">
333 <h1>🔁 UPDATE</h1>
334 </header>

339 <tr><th>TYPE</th><th>VALUE</th></tr>
340 </thead>
341 <tbody>
342 {currencies.map((c) => (
343 <tr key={c.type}>
344 <td>{c.type}</td>
345 <td>{c.value}</td>
346 </tr>
347 ))}
348 </tbody>
349 </table>
