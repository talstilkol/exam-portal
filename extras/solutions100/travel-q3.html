1// Travel-SV — Node Q3 (25 pts)
2// "חבילת Node — מחלקת קובץ טקסט"
3// A class with read/write/append/delete methods using fs/promises (no callbacks).
4// Atomic write via temp file + rename. ENOENT handling.
5 
6const fs = require("fs/promises");
7const path = require("path");
8const os = require("os");
9 
10class TextFile {
11 /**
12 * @param {string} filePath — absolute or relative path
13 */
14 constructor(filePath) {
15 if (typeof filePath !== "string" || !filePath) {
16 throw new TypeError("filePath must be a non-empty string");
17 }
18 this.filePath = path.resolve(filePath);
19 }
20 
21 /**
22 * Read full content. Returns '' if file does not exist (ENOENT).
23 * @returns {Promise<string>}
24 */
25 async read() {
26 try {
27 // ✅ FIXED: async/await (no callbacks)
28 return await fs.readFile(this.filePath, "utf-8");
29 } catch (err) {
30 // ✅ FIXED: ENOENT → return empty string instead of throwing
31 if (err.code === "ENOENT") return "";
32 throw err;
33 }
34 }
35 
36 /**
37 * Returns the file as an array of lines (without trailing empty).
38 * @returns {Promise<string[]>}
39 */
40 async readLines() {
41 const content = await this.read();
42 if (!content) return [];
43 return content.split(/\r?\n/).filter((l) => l.length > 0);
44 }
45 
46 /**
47 * Atomic write: write to temp file in same dir, then rename.
48 * If anything fails midway, the original file is untouched.
49 * @param {string} content
50 */
51 async write(content) {
52 if (typeof content !== "string") {
53 throw new TypeError("content must be a string");
54 }
55 const dir = path.dirname(this.filePath);
56 // Ensure parent dir exists
57 await fs.mkdir(dir, { recursive: true });
58 const tmpName = `.${path.basename(this.filePath)}.${process.pid}.${Date.now()}.tmp`;
59 const tmpPath = path.join(dir, tmpName);
60 try {
61 // ✅ Atomic write: write to temp, then rename (fs.rename is atomic on same fs)
62 await fs.writeFile(tmpPath, content, "utf-8");
63 await fs.rename(tmpPath, this.filePath);
64 } catch (err) {
65 // Cleanup tmp on failure
66 try { await fs.unlink(tmpPath); } catch {}
67 throw err;
68 }
69 }
70 
71 /**
72 * Append content (creates file if missing).
73 * @param {string} content
74 */
75 async append(content) {
76 if (typeof content !== "string") {
77 throw new TypeError("content must be a string");
78 }
79 const dir = path.dirname(this.filePath);
80 await fs.mkdir(dir, { recursive: true });
81 await fs.appendFile(this.filePath, content, "utf-8");
82 }
83 
84 /**
85 * Append a line (auto-adds \n if missing).
86 */
87 async appendLine(line) {
88 const text = line.endsWith("\n") ? line : line + "\n";
89 await this.append(text);
90 }
91 
92 /**
93 * Delete the file. ENOENT is treated as success (already gone).
94 */
95 async delete() {
96 try {
97 await fs.unlink(this.filePath);
98 } catch (err) {
99 if (err.code === "ENOENT") return;
100 throw err;
101 }
102 }
103 
104 /**
105 * Returns true if the file exists and is readable.
106 */
107 async exists() {
108 try {
109 await fs.access(this.filePath, fs.constants.F_OK);
110 return true;
111 } catch { return false; }
112 }
113 
114 /**
115 * File size in bytes (0 if not found).
116 */
117 async size() {
118 try {
119 const stat = await fs.stat(this.filePath);
120 return stat.size;
121 } catch (err) {
122 if (err.code === "ENOENT") return 0;
123 throw err;
124 }
125 }
126}
