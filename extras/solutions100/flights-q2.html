4/**
5 * @param {number[][]} matrix
6 * @param {number} target
7 * @returns {{count:number, positions:Array<{row:number, col:number}>}}
8 */
9function countMatrix(matrix, target) {
10 if (!Array.isArray(matrix)) {
11 throw new TypeError(`expected a matrix (2D array), got ${typeof matrix}`);
12 }
13 if (typeof target !== "number" || !Number.isFinite(target)) {
14 throw new TypeError(`target must be a finite number`);
15 }
16 
17 let count = 0;
18 const positions = [];
19 
20 for (let r = 0; r < matrix.length; r++) {
21 const row = matrix[r];
22 if (!Array.isArray(row)) {
23 throw new TypeError(`row ${r} is not an array`);
24 }
25 for (let c = 0; c < row.length; c++) {
26 const v = row[c];
27 // ✅ FIXED: throw on non-numeric values per spec
28 if (typeof v !== "number" || !Number.isFinite(v)) {
29 throw new TypeError(`non-number at [${r}][${c}]: ${JSON.stringify(v)}`);
30 }
31 if (v === target) {
32 count++;
33 positions.push({ row: r, col: c });
34 }
35 }
36 }
37 return { count, positions };
38}
