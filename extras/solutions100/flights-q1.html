59// ───────────────────── Page 1 — Flight list ─────────────────────
60function FlightList({ flights, setFlights }) {
61 const [search, setSearch] = useState("");
62 const [statusFilter, setStatusFilter] = useState("all");
63 const navigate = useNavigate();
64 
65 // ✅ Debounced search via simple effect
66 const [debounced, setDebounced] = useState(search);
67 useEffect(() => {
68 const t = setTimeout(() => setDebounced(search), 200);
69 return () => clearTimeout(t);
70 }, [search]);
71 
72 const filtered = useMemo(() => {
73 const term = debounced.trim().toUpperCase();
74 return flights.filter((f) => {
75 if (statusFilter !== "all" && f.status !== statusFilter) return false;
76 if (!term) return true;
77 return (
78 f.number.includes(term) ||
79 f.from.includes(term) ||
80 f.to.includes(term)
81 );
82 });
83 }, [flights, debounced, statusFilter]);
84 
85 function updateStatus(id, newStatus) {
86 const flight = flights.find((f) => f.id === id);
87 if (!flight) return;
88 // ✅ FIXED: skip alert if same status (per spec)
89 if (flight.status === newStatus) return;
90 setFlights(flights.map((f) => f.id === id ? { ...f, status: newStatus } : f));
91 alert(`Flight ${flight.number}: ${flight.status} → ${newStatus}`);
92 }
93 
94 function remove(id) {
95 if (!confirm("Delete this flight?")) return;
96 setFlights(flights.filter((f) => f.id !== id));
97 }
98 
99 return (
100 <main className="page">
101 <header className="hero">
102 <h1>✈️ Flight Control</h1>
103 <p>{filtered.length} flight(s)</p>
104 </header>
105 <section className="controls">
106 <input
107 type="search"
108 placeholder="Search flight # / from / to"
109 value={search}
110 onChange={(e) => setSearch(e.target.value.toUpperCase())}
111 />
112 <select value={statusFilter} onChange={(e) => setStatusFilter(e.target.value)}>
113 <option value="all">All statuses</option>

145// ───────────────────── Page 2 — Flight form (add/edit) ─────────────────────
146function FlightForm({ flights, setFlights, mode }) {
147 const navigate = useNavigate();
148 // useParams not strictly needed here since we use window.location for simplicity
149 const id = mode === "edit"
150 ? Number(window.location.pathname.split("/").pop())
151 : null;
152 const existing = mode === "edit" ? flights.find((f) => f.id === id) : null;
153 
154 const [form, setForm] = useState(existing || {
155 number: "", from: "", to: "", status: "scheduled", time: "",
156 });
157 const [errors, setErrors] = useState({});
158 
159 function setField(field, value) {
160 setForm((f) => ({ ...f, [field]: value }));
161 if (errors[field]) {
162 setErrors((e) => { const c = { ...e }; delete c[field]; return c; });
163 }
164 }
165 
166 function submit(e) {
167 e.preventDefault();
168 const errs = {};
169 // ✅ Validation: flight number regex
170 if (!FLIGHT_NUMBER_RE.test(form.number))
171 errs.number = "Flight number must be 2 letters + 3-4 digits (e.g. LY007)";
172 // ✅ Validation: airport codes 3 uppercase letters
173 if (!AIRPORT_CODE_RE.test(form.from)) errs.from = "From: 3 uppercase letters";
174 if (!AIRPORT_CODE_RE.test(form.to)) errs.to = "To: 3 uppercase letters";
175 if (form.from === form.to) errs.to = "Origin and destination must differ";
176 if (!form.time) errs.time = "Departure time is required";
177 if (Object.keys(errs).length > 0) { setErrors(errs); return; }
178 
179 if (mode === "add") {
180 setFlights([...flights, { ...form, id: Date.now() }]);
181 } else {
182 setFlights(flights.map((f) => f.id === id ? { ...form, id } : f));
183 }
184 navigate("/");
185 }
186 
187 return (
188 <main className="page center">
189 <h1>{mode === "add" ? "✈️ Add Flight" : "✏️ Edit Flight"}</h1>
190 <form onSubmit={submit} className="flight-form" noValidate>
191 <Field label="Flight number (e.g. LY007)" value={form.number} onChange={(v) => setField("number", v.toUpperCase())} error={errors.number} />
192 <Field label="From (IATA code)" value={form.from} onChange={(v) => setField("from", v.toUpperCase())} error={errors.from} maxLength={3} />
193 <Field label="To (IATA code)" value={form.to} onChange={(v) => setField("to", v.toUpperCase())} error={errors.to} maxLength={3} />
194 <label>
195 Status:
196 <select value={form.status} onChange={(e) => setField("status", e.target.value)}>
197 {STATUSES.map((s) => <option key={s.value} value={s.value}>{s.label}</option>)}
198 </select>
199 </label>
200 <label>
201 Departure:
202 <input type="datetime-local" value={form.time}
203 onChange={(e) => setField("time", e.target.value)}
204 aria-invalid={!!errors.time} />
205 {errors.time && <p className="err-text err-red">{errors.time}</p>}
206 </label>
207 <div className="row">
208 <button type="submit" className="btn-primary">{mode === "add" ? "Add" : "Save"}</button>
209 <button type="button" className="btn-back" onClick={() => navigate("/")}>← Back</button>
210 </div>
211 </form>
212 </main>
213 );
214}

216function Field({ label, value, onChange, error, ...rest }) {
217 return (
218 <div className={`field ${error ? "has-error" : ""}`}>
219 {error && <p className="err-text err-red" role="alert">{error}</p>}
220 <input type="text" placeholder={label} value={value}
221 onChange={(e) => onChange(e.target.value)}
222 aria-label={label} aria-invalid={!!error} {...rest} />
223 </div>
224 );
