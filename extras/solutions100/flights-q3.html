1// Flights — Server Q3 (25 pts)
2// "שרת Express + MongoDB — סטודנטים"
3// Endpoints:
4// GET /students → all students
5// GET /students/:id → single student
6// POST /students → create student (id must be unique)
7// PUT /students/:id → update existing
8// DELETE /students/:id → remove
9// POST /students/filter→ filter by gpa ≥ X / year / name substring
10 
11const express = require("express");
12const mongoose = require("mongoose");
13 
14const PORT = process.env.PORT || 3003;
15const MONGO_URL = process.env.MONGO_URL || "mongodb://127.0.0.1:27017/flights";
16 
17const studentSchema = new mongoose.Schema(
18 {
19 id: { type: Number, required: true, unique: true, index: true },
20 name: { type: String, required: true, trim: true },
21 age: { type: Number, required: true, min: 18 },
22 gpa: { type: Number, required: true, min: 0, max: 100 },
23 year: { type: Number, required: true, min: 1, max: 6 },
24 email: { type: String, trim: true, lowercase: true },
25 },
26 { timestamps: true }
27);
28const Student = mongoose.model("Student", studentSchema);
29 
30const app = express();
31app.use(express.json());
32 
33function validateStudent(body) {
34 const errs = [];
35 if (typeof body.id !== "number") errs.push("id (number) required");
36 if (typeof body.name !== "string" || !body.name.trim()) errs.push("name required");
37 if (typeof body.age !== "number" || body.age < 18) errs.push("age must be ≥ 18");
38 if (typeof body.gpa !== "number" || body.gpa < 0 || body.gpa > 100) errs.push("gpa must be 0–100");
39 if (typeof body.year !== "number" || body.year < 1 || body.year > 6) errs.push("year must be 1–6");
40 return errs;
41}
42 
43// GET /students
44app.get("/students", async (_req, res) => {
45 try {
46 const all = await Student.find().lean();
47 res.json(all);
48 } catch (e) { res.status(500).json({ error: e.message }); }
49});
50 
51// GET /students/:id
52app.get("/students/:id", async (req, res) => {
53 try {
54 const s = await Student.findOne({ id: Number(req.params.id) }).lean();
55 if (!s) return res.status(404).json({ error: "Student not found" });
56 res.json(s);
57 } catch (e) { res.status(500).json({ error: e.message }); }
58});
59 
60// POST /students — create with id duplicate check
61app.post("/students", async (req, res) => {
62 const errs = validateStudent(req.body);
63 if (errs.length > 0) return res.status(400).json({ error: "Invalid body", errors: errs });
64 try {
65 // ✅ FIXED: id duplicate check before insert
66 const exists = await Student.findOne({ id: req.body.id });
67 if (exists) return res.status(409).json({ error: `Student with id ${req.body.id} already exists` });
68 const created = await Student.create(req.body);
69 res.status(201).json(created);
70 } catch (e) {
71 if (e.code === 11000) return res.status(409).json({ error: "Duplicate id" });
72 res.status(500).json({ error: e.message });
73 }
74});
75 
76// PUT /students/:id — update existing
77app.put("/students/:id", async (req, res) => {
78 try {
79 const updated = await Student.findOneAndUpdate(
80 { id: Number(req.params.id) },
81 req.body,
82 { new: true, runValidators: true }
83 );
84 if (!updated) return res.status(404).json({ error: "Student not found" });
85 res.json(updated);
86 } catch (e) { res.status(500).json({ error: e.message }); }
87});
88 
89// DELETE /students/:id
90app.delete("/students/:id", async (req, res) => {
91 try {
92 const removed = await Student.findOneAndDelete({ id: Number(req.params.id) });
93 if (!removed) return res.status(404).json({ error: "Student not found" });
94 res.status(204).end();
95 } catch (e) { res.status(500).json({ error: e.message }); }
96});
97 
98// POST /students/filter — combined filters
99app.post("/students/filter", async (req, res) => {
100 const { minGpa, year, nameContains } = req.body || {};
101 const query = {};
102 if (typeof minGpa === "number") query.gpa = { $gte: minGpa };
103 if (typeof year === "number") query.year = year;
104 if (typeof nameContains === "string" && nameContains.trim())
105 query.name = { $regex: nameContains.trim(), $options: "i" };
106 try {
107 const list = await Student.find(query).lean();
108 res.json(list);
109 } catch (e) { res.status(500).json({ error: e.message }); }
110});
111 
112// Error middleware
113app.use((err, _req, res, _next) => {
114 console.error(err);
115 res.status(500).json({ error: "Server error", message: err.message });
116});
117 
118(async () => {
119 try {
120 await mongoose.connect(MONGO_URL);
121 app.listen(PORT, () => console.log(`Flights Q3 on http://localhost:${PORT}`));
122 } catch (err) {
123 console.error("Boot failed:", err);
124 process.exit(1);
125 }
126})();
