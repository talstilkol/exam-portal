72// ───────────────────── Page 2 — Add Event ─────────────────────
73function AddEvent({ events, setEvents }) {
74 const [form, setForm] = useState({ title: "", city: "", category: CATEGORIES[0], phone: "", description: "" });
75 const [errors, setErrors] = useState({});
76 const navigate = useNavigate();
77 
78 function setField(field, value) {
79 setForm((f) => ({ ...f, [field]: value }));
80 if (errors[field]) setErrors((e) => { const c = { ...e }; delete c[field]; return c; });
81 }
82 
83 function submit(e) {
84 e.preventDefault();
85 const errs = {};
86 // ✅ Validation: city required
87 if (!form.city.trim()) errs.city = "City is required";
88 // ✅ Validation: title required
89 if (!form.title.trim()) errs.title = "Title is required";
90 // ✅ Validation: phone 9-10 digits (Israeli format)
91 if (!/^\d{9,10}$/.test(form.phone)) errs.phone = "Phone must be 9-10 digits";
92 if (Object.keys(errs).length > 0) { setErrors(errs); return; }
93 
94 setEvents([...events, { ...form, id: Date.now(), selected: false }]);
95 alert("Event added!");
96 navigate("/");
97 }
98 
99 return (
100 <main className="page center">
101 <h1>הוספת התנדבות</h1>
102 <form onSubmit={submit} className="add-form" noValidate>
103 <Field label="Title" value={form.title} onChange={(v) => setField("title", v)} error={errors.title} />
104 <Field label="City" value={form.city} onChange={(v) => setField("city", v)} error={errors.city} />
105 <label>
106 Category:
107 <select value={form.category} onChange={(e) => setField("category", e.target.value)}>
108 {CATEGORIES.map((c) => <option key={c} value={c}>{c}</option>)}
109 </select>
110 </label>
111 <Field label="Phone (9-10 digits)" value={form.phone} onChange={(v) => setField("phone", v)} error={errors.phone} inputMode="numeric" />
112 <label>
113 Description:
114 <textarea rows="3" value={form.description} onChange={(e) => setField("description", e.target.value)} />
115 </label>
116 <div className="row">
117 <button type="submit" className="btn-primary">Add</button>
118 <button type="button" className="btn-back" onClick={() => navigate("/")}>← Back</button>
119 </div>
120 </form>
121 </main>
122 );
123}

125function Field({ label, value, onChange, error, ...rest }) {
126 return (
127 <div className={`field ${error ? "has-error" : ""}`}>
128 {error && <p className="err-text err-red" role="alert">{error}</p>}
129 <input type="text" placeholder={label} value={value}
130 onChange={(e) => onChange(e.target.value)}
131 aria-label={label} aria-invalid={!!error} {...rest} />
132 </div>
133 );
134}

136// ───────────────────── Page 3 — Find Events ─────────────────────
137function FindEvents({ events, setEvents }) {
138 const [city, setCity] = useState("all");
139 const [category, setCategory] = useState("all");
140 const navigate = useNavigate();
141 
142 const cities = useMemo(() => Array.from(new Set(events.map((e) => e.city))).sort(), [events]);
143 
144 const filtered = useMemo(() => {
145 return events.filter((e) => {
146 if (city !== "all" && e.city !== city) return false;
147 if (category !== "all" && e.category !== category) return false;
148 return true;
149 });
150 }, [events, city, category]);
151 
152 function toggleSelect(id) {
153 setEvents(events.map((e) => e.id === id ? { ...e, selected: !e.selected } : e));
154 }
155 
156 return (
157 <main className="page">
158 <header className="hero">
159 <h1>🔍 מצא התנדבות</h1>
160 <p>{filtered.length} opportunit{filtered.length === 1 ? "y" : "ies"}</p>
161 </header>
162 <section className="filter-bar">
163 <select value={city} onChange={(e) => setCity(e.target.value)} aria-label="Filter by city">
164 <option value="all">All cities</option>
165 {cities.map((c) => <option key={c} value={c}>{c}</option>)}
166 </select>
167 <select value={category} onChange={(e) => setCategory(e.target.value)} aria-label="Filter by category">
168 <option value="all">All categories</option>
169 {CATEGORIES.map((c) => <option key={c} value={c}>{c}</option>)}
170 </select>
171 </section>
172 <section className="events-grid">
173 {filtered.length === 0 && <p className="empty">No opportunities matching</p>}
174 {filtered.map((e) => (
175 <article key={e.id} className={`event-card ${e.selected ? "is-selected" : ""}`}>
176 <header>
177 <h3>{e.title}</h3>
178 <span className="cat-badge">{e.category}</span>
179 </header>
180 <p>📍 {e.city}</p>
181 <p>📞 {e.phone}</p>
182 <p>{e.description}</p>
183 <button className={e.selected ? "btn-secondary" : "btn-primary"}
184 onClick={() => toggleSelect(e.id)}
185 aria-pressed={e.selected}>
186 {e.selected ? "✓ Selected" : "Select"}
187 </button>
188 </article>
189 ))}
190 </section>
191 <button className="btn-back" onClick={() => navigate("/")}>← Back</button>
192 </main>
193 );
