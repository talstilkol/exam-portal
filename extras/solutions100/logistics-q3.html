  • ✅ POST /shipments מחזיר 200 במקום 201 — תוקן ל־status(201)
  • ✅ PUT ו־DELETE לא בודקים 404 כשהמשלוח לא נמצא — נוספה בדיקת קיום

1// Logistics Q3 — Shipment Management API (25 pts)
2// GET /shipments → all shipments
3// POST /shipments → add shipment (validated)
4// PUT /shipments/:id/status → update status
5// DELETE /shipments/:id → remove shipment
6// GET /shipments/heavy?min=X → filter by weight
7 
8const express = require("express");
9const cors = require("cors");
10 
11const app = express();
12app.use(express.json());
13app.use(cors());
14 
15let shipments = [];
16let nextId = 1;
17 
18// GET /shipments — return all shipments
19app.get("/shipments", (req, res) => {
20 res.json(shipments);
21});
22 
23// GET /shipments/heavy?min=X — filter by weight >= min
24app.get("/shipments/heavy", (req, res) => {
25 const min = Number(req.query.min) || 0;
26 const heavy = shipments.filter(s => s.weight >= min);
27 res.json(heavy);
28});
29 
30// POST /shipments — add a new shipment
31app.post("/shipments", (req, res) => {
32 const { sender, receiver, weight, destination } = req.body;
33 
34 if (!sender || !receiver || !weight || !destination) {
35 return res.status(400).json({ error: "All fields required: sender, receiver, weight, destination" });
36 }
37 if (typeof weight !== "number" || weight <= 0) {
38 return res.status(400).json({ error: "weight must be a positive number" });
39 }
40 
41 const shipment = {
42 id: nextId++,
43 sender,
44 receiver,
45 weight,
46 destination,
47 status: "pending"
48 };
49 shipments.push(shipment);
50 res.status(201).json(shipment);
51});
52 
53// PUT /shipments/:id/status — update shipment status
54app.put("/shipments/:id/status", (req, res) => {
55 const shipment = shipments.find(s => s.id === Number(req.params.id));
56 if (!shipment) {
57 return res.status(404).json({ error: "Shipment not found" });
58 }
59 
60 const { status } = req.body;
61 const allowed = ["pending", "shipped", "delivered"];
62 if (!allowed.includes(status)) {
63 return res.status(400).json({ error: "status must be: pending, shipped, or delivered" });
64 }
65 
66 shipment.status = status;
67 res.json(shipment);
68});
69 
70// DELETE /shipments/:id — remove a shipment
71app.delete("/shipments/:id", (req, res) => {
72 const idx = shipments.findIndex(s => s.id === Number(req.params.id));
73 if (idx === -1) {
74 return res.status(404).json({ error: "Shipment not found" });
75 }
76 shipments.splice(idx, 1);
77 res.status(204).send();
78});
79 
80// Start server
81const PORT = 3000;
82app.listen(PORT, () => {
83 console.log(`Logistics server running on http://localhost:${PORT}`);
84});
