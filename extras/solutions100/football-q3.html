1// Football Club — Express Q3 (25 pts)
2// Computer-game catalog API
3// GET / → all games
4// POST / → add a game (validated)
5// POST /search → filter by yearMin / nameContains / priceMax (any combo, max 3)
6//
7// Each game: { name: string, description: string, releaseYear: number, price: number }
8 
9const express = require("express");
10const fs = require("fs/promises");
11 
12const PORT = process.env.PORT || 3001;
13const DB_FILE = "./games.json";
14 
15// ── persistence helpers ──────────────────────────────────────────────
16async function loadGames() {
17 try {
18 const raw = await fs.readFile(DB_FILE, "utf-8");
19 return JSON.parse(raw);
20 } catch (err) {
21 if (err.code === "ENOENT") return [];
22 throw err;
23 }
24}
25async function saveGames(games) {
26 await fs.writeFile(DB_FILE, JSON.stringify(games, null, 2));
27}
28 
29// ── input validation ────────────────────────────────────────────────
30function validateGame(body) {
31 const errs = [];
32 if (typeof body.name !== "string" || !body.name.trim())
33 errs.push("name is required (non-empty string)");
34 if (typeof body.description !== "string")
35 errs.push("description must be a string");
36 if (!Number.isInteger(body.releaseYear) ||
37 body.releaseYear < 1950 || body.releaseYear > 2100)
38 errs.push("releaseYear must be an integer between 1950 and 2100");
39 if (typeof body.price !== "number" || body.price < 0)
40 errs.push("price must be a non-negative number");
41 return errs;
42}
43 
44// ── server boot ────────────────────────────────────────────────────
45const app = express();
46app.use(express.json());
47 
48// log every request (helpful when debugging at the exam)
49app.use((req, _res, next) => {
50 console.log(`${req.method} ${req.url}`);
51 next();
52});
53 
54// in-memory cache; loaded at boot
55let games = [];
56 
57// GET / — return all games
58app.get("/", async (_req, res) => {
59 try {
60 res.json(games);
61 } catch (err) {
62 res.status(500).json({ error: "Failed to read games" });
63 }
64});
65 
66// POST / — add a new game
67app.post("/", async (req, res) => {
68 const errors = validateGame(req.body);
69 if (errors.length > 0) {
70 return res.status(400).json({ error: "Invalid body", errors });
71 }
72 const game = {
73 id: Date.now().toString(36),
74 name: req.body.name.trim(),
75 description: req.body.description,
76 releaseYear: req.body.releaseYear,
77 price: req.body.price,
78 };
79 games.push(game);
80 await saveGames(games);
81 res.status(201).json(game);
82});
83 
84// POST /search — filter by any combination of up to 3 fields
85// { yearMin?: number, nameContains?: string, priceMax?: number }
86// ✅ FIXED: combined AND filter so all 3 work together (per spec)
87app.post("/search", (req, res) => {
88 const { yearMin, nameContains, priceMax } = req.body || {};
89 const filtered = games.filter((g) => {
90 if (typeof yearMin === "number" && g.releaseYear < yearMin) return false;
91 if (typeof nameContains === "string" && nameContains &&
92 !g.name.toLowerCase().includes(nameContains.toLowerCase())) return false;
93 if (typeof priceMax === "number" && g.price > priceMax) return false;
94 return true;
95 });
96 res.json(filtered);
97});
98 
99// global error middleware
100app.use((err, _req, res, _next) => {
101 console.error(err);
102 res.status(500).json({ error: "Server error", message: err.message });
103});
104 
105(async () => {
106 games = await loadGames();
107 app.listen(PORT, () => console.log(`Football Q3 server on http://localhost:${PORT}`));
108})();
