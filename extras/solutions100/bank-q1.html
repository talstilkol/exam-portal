200// ───────────────────── Transfer (with confirm modal) ─────────────────────
201function TransferPage({ accounts, setAccounts, record }) {
202 const [fromId, setFromId] = useState(accounts[0]?.id || "");
203 const [toId, setToId] = useState(accounts[1]?.id || "");
204 const [amount, setAmount] = useState("");
205 const [error, setError] = useState("");
206 const [confirming, setConfirming] = useState(null);
207 const navigate = useNavigate();
208 
209 function attempt(e) {
210 e.preventDefault();
211 setError("");
212 const n = Number(amount);
213 if (!Number.isFinite(n) || n <= 0) { setError("Amount must be a positive number"); return; }
214 if (Number(fromId) === Number(toId)) {
215 // ✅ FIXED: source ≠ target check
216 setError("Source and target accounts must be different");
217 return;
218 }
219 const src = accounts.find((a) => a.id === Number(fromId));
220 const dst = accounts.find((a) => a.id === Number(toId));
221 if (!src || !dst) { setError("One of the accounts was not found"); return; }
222 if (n > src.balance) { setError(`Insufficient funds. Available: ₪${src.balance}`); return; }
223 
224 // ✅ Confirmation modal for large transfers
225 if (n > TRANSFER_CONFIRM_THRESHOLD) {
226 setConfirming({ src, dst, amount: n });
227 return;
228 }
229 execute(src, dst, n);
230 }
231 
232 function execute(src, dst, n) {
233 const newSrcBalance = src.balance - n;
234 const newDstBalance = dst.balance + n;
235 setAccounts(accounts.map((a) => {
236 if (a.id === src.id) return { ...a, balance: newSrcBalance };
237 if (a.id === dst.id) return { ...a, balance: newDstBalance };
238 return a;
239 }));
240 record("transfer", src.id, dst.id, n, newSrcBalance, `→ ${dst.owner}`);
241 alert(`Transferred ₪${n} from ${src.owner} to ${dst.owner}`);
242 navigate("/");
243 }
244 
245 return (
246 <main className="page">
247 <h1>🔄 Transfer</h1>
248 <form onSubmit={attempt} className="form-card">
249 <label>
250 From:
251 <select value={fromId} onChange={(e) => { setFromId(e.target.value); setError(""); }}>
252 {accounts.map((a) => (
253 <option key={a.id} value={a.id}>#{a.id} — {a.owner} (₪{a.balance})</option>
254 ))}
255 </select>
256 </label>
257 <label>
258 To:
259 <select value={toId} onChange={(e) => { setToId(e.target.value); setError(""); }}>
260 {accounts.map((a) => (
261 <option key={a.id} value={a.id}>#{a.id} — {a.owner} (₪{a.balance})</option>
262 ))}
263 </select>
264 </label>
265 <label>
266 Amount:
267 <input type="number" min="0.01" step="0.01"
268 value={amount} onChange={(e) => { setAmount(e.target.value); setError(""); }} />
269 </label>
270 {error && <p className="err-text err-red">{error}</p>}
271 <button type="submit" className="btn-primary">Transfer</button>
272 <button type="button" className="btn-back" onClick={() => navigate(-1)}>← Back</button>
273 </form>
274 
275 {/* ✅ Confirmation modal for large amounts */}
276 {confirming && (
277 <div className="modal-backdrop" role="dialog" aria-modal="true" onClick={() => setConfirming(null)}>
278 <div className="modal" onClick={(e) => e.stopPropagation()}>
279 <h2>Confirm large transfer</h2>
280 <p>You are about to transfer <strong>₪{confirming.amount}</strong> from
281 <br/>{confirming.src.owner} → {confirming.dst.owner}</p>
282 <p>This is above the ₪{TRANSFER_CONFIRM_THRESHOLD} threshold.</p>
283 <div className="modal-actions">
284 <button onClick={() => { execute(confirming.src, confirming.dst, confirming.amount); setConfirming(null); }} className="btn-primary">Confirm</button>
285 <button onClick={() => setConfirming(null)} className="btn-back">Cancel</button>
286 </div>
287 </div>
288 </div>
289 )}
290 </main>
291 );
292}

294// ───────────────────── History (search/filter) ─────────────────────
295function HistoryPage({ history, accounts }) {
296 const [filter, setFilter] = useState("all");
297 const [search, setSearch] = useState("");
298 const navigate = useNavigate();
299 
300 const filtered = useMemo(() => {
301 return history
302 .slice()
303 .reverse()
304 .filter((h) => filter === "all" || h.type === filter)
305 .filter((h) => {
306 if (!search) return true;
307 const term = search.toLowerCase();
308 const fromAcc = accounts.find((a) => a.id === h.fromId);
309 const toAcc = accounts.find((a) => a.id === h.toId);
310 return (
311 (fromAcc && fromAcc.owner.toLowerCase().includes(term)) ||
312 (toAcc && toAcc.owner.toLowerCase().includes(term)) ||
313 String(h.amount).includes(term)
314 );
315 });
316 }, [history, filter, search, accounts]);
317 
318 return (
319 <main className="page">
320 <h1>📋 Transaction History</h1>
321 <div className="history-controls">
322 <select value={filter} onChange={(e) => setFilter(e.target.value)}>
323 <option value="all">All</option>
324 <option value="deposit">Deposits</option>
325 <option value="withdraw">Withdrawals</option>
326 <option value="transfer">Transfers</option>
327 </select>
328 <input
329 type="search" placeholder="Search owner / amount"
330 value={search} onChange={(e) => setSearch(e.target.value)}
331 />
332 </div>
333 <table className="history-table">
334 <thead>
335 <tr><th>Time</th><th>Type</th><th>From</th><th>To</th><th>Amount</th><th>Balance after</th></tr>
336 </thead>
337 <tbody>
338 {filtered.map((h) => {
339 const src = accounts.find((a) => a.id === h.fromId);
340 const dst = accounts.find((a) => a.id === h.toId);
341 return (
342 <tr key={h.id}>
343 <td>{new Date(h.timestamp).toLocaleString("he-IL")}</td>
344 <td><span className={`tag tag-${h.type}`}>{h.type}</span></td>
345 <td>{src ? src.owner : "—"}</td>
346 <td>{dst ? dst.owner : "—"}</td>
347 <td>₪{h.amount.toLocaleString()}</td>
348 <td>₪{h.balanceAfter.toLocaleString()}</td>
349 </tr>
350 );
351 })}
352 </tbody>
353 </table>
354 {filtered.length === 0 && <p className="empty">No matching transactions</p>}
355 <button className="btn-back" onClick={() => navigate("/")}>← Back</button>
356 </main>
357 );
