25// ───────────────────── Persistence ─────────────────────
26const KEY_USERS = "parking-users";
27const KEY_SESSION = "parking-session";
28const KEY_ACTIVE = "parking-active";
29const KEY_HISTORY = "parking-history";

31function loadUsers() { try { const r = localStorage.getItem(KEY_USERS); return r ? JSON.parse(r) : []; } catch { return []; } }
32function saveUsers(arr) { try { localStorage.setItem(KEY_USERS, JSON.stringify(arr)); } catch {} }
33function loadSession() { try { return localStorage.getItem(KEY_SESSION); } catch { return null; } }
34function saveSession(u) { try { u ? localStorage.setItem(KEY_SESSION, u) : localStorage.removeItem(KEY_SESSION); } catch {} }
35function loadActive() { try { const r = localStorage.getItem(KEY_ACTIVE); return r ? JSON.parse(r) : null; } catch { return null; } }
36function saveActive(a) { try { a ? localStorage.setItem(KEY_ACTIVE, JSON.stringify(a)) : localStorage.removeItem(KEY_ACTIVE); } catch {} }
37function loadHistory() { try { const r = localStorage.getItem(KEY_HISTORY); return r ? JSON.parse(r) : []; } catch { return []; } }
38function saveHistory(h) { try { localStorage.setItem(KEY_HISTORY, JSON.stringify(h)); } catch {} }

82// ───────────────────── Page 2 — Sign Up ─────────────────────
83function Signup({ users, setUsers, setSession }) {
84 const [form, setForm] = useState({ username: "", password: "", car: "", phone: "" });
85 const [errors, setErrors] = useState({});
86 const navigate = useNavigate();
87 
88 function setField(field, value) {
89 setForm((f) => ({ ...f, [field]: value }));
90 if (errors[field]) {
91 setErrors((e) => { const c = { ...e }; delete c[field]; return c; });
92 }
93 }
94 
95 function submit(e) {
96 e.preventDefault();
97 const errs = {};
98 if (!form.username.trim()) errs.username = "Username is required";
99 if (users.some((u) => u.username === form.username)) errs.username = "Username already exists";
100 if (form.password.length < 6) errs.password = "Password must be at least 6 chars";
101 if (!form.car.trim()) errs.car = "Car number is required";
102 // ✅ Phone: 9-10 digits per Israeli format
103 if (!/^\d{9,10}$/.test(form.phone)) errs.phone = "Phone must be 9-10 digits";
104 if (Object.keys(errs).length > 0) { setErrors(errs); return; }
105 setUsers([...users, { ...form }]);
106 setSession(form.username);
107 navigate("/choose-parking", { replace: true });
108 }
109 
110 return (
111 <main className="page center">
112 <h1>Sign Up</h1>
113 <form onSubmit={submit} className="signup-form" noValidate>
114 <Field label="Username" value={form.username} onChange={(v) => setField("username", v)} error={errors.username} />
115 <Field label="Password" type="password" value={form.password} onChange={(v) => setField("password", v)} error={errors.password} />
116 <Field label="Car number" value={form.car} onChange={(v) => setField("car", v)} error={errors.car} />
117 <Field label="Phone (9-10 digits)" value={form.phone} onChange={(v) => setField("phone", v)} error={errors.phone} inputMode="numeric" />
118 <button type="submit" className="btn-primary">Sign Up</button>
119 </form>
120 </main>
121 );
122}

124// Reusable field
125function Field({ label, value, onChange, error, type = "text", ...rest }) {
126 return (
127 <div className={`field ${error ? "has-error" : ""}`}>
128 {error && <p className="err-text err-red" role="alert">{error}</p>}
129 <input
130 type={type}
131 placeholder={label}
132 value={value}
133 onChange={(e) => onChange(e.target.value)}
134 aria-label={label}
135 aria-invalid={!!error}
136 {...rest}
137 />
138 </div>
139 );
140}

142// ───────────────────── Page 3 — Sign In ─────────────────────
143function Signin({ users, setSession }) {
144 const [form, setForm] = useState({ username: "", password: "" });

146 const navigate = useNavigate();

149 const found = users.find((u) => u.username === form.username && u.password === form.password);

151 setSession(form.username);
152 navigate("/choose-parking", { replace: true });
153 }
154 return (
155 <main className="page center">
156 <h1>Sign In</h1>
157 <form onSubmit={submit} className="signin-form" noValidate>
158 <input type="text" placeholder="Username"
159 value={form.username} onChange={(e) => setForm({ ...form, username: e.target.value })} />
160 <input type="password" placeholder="Password"
161 value={form.password} onChange={(e) => setForm({ ...form, password: e.target.value })} />
162 {error && <p className="err-text err-red">{error}</p>}
163 <button type="submit" className="btn-primary">Sign In</button>
164 </form>

187// ───────────────────── Page 4 — Choose Parking ─────────────────────
188function Choose({ users, session, active, setActive, setSession }) {
189 const me = users.find((u) => u.username === session);
190 const [city, setCity] = useState("");
191 const [error, setError] = useState("");
192 const navigate = useNavigate();
193 
194 function start() {
195 setError("");
196 // ✅ FIXED: city + car required
197 if (!city) { setError("Please choose a city"); return; }
198 if (!me?.car) { setError("Car number missing on profile"); return; }
199 if (active) { setError("You already have an active parking. Pay first."); return; }
200 setActive({
201 city,
202 pricePerHour: CITY_PRICES[city],
203 car: me.car,
204 startedAt: new Date().toISOString(),
205 });
206 navigate("/active-parking");
207 }
208 
209 return (
210 <Layout session={session} setSession={setSession}>
211 <h1>Choose Parking</h1>
212 {active && (
213 <div className="alert">
214 You have an active parking in <strong>{active.city}</strong>.
215 <Link to="/active-parking">Go to active →</Link>
216 </div>
217 )}
218 <div className="city-grid">
219 {Object.entries(CITY_PRICES).map(([name, price]) => (
220 <button
221 key={name}
222 className={`city-card ${city === name ? "selected" : ""}`}
223 onClick={() => setCity(name)}
224 aria-pressed={city === name}
225 >
226 <h3>{name}</h3>
227 {/* ✅ Exact prices per spec */}
228 <p className="price">₪{price}/h</p>
229 </button>
230 ))}
231 </div>
232 <p>Car: <strong>{me?.car || "—"}</strong></p>

234 <button className="btn-primary" onClick={start} disabled={!city || !!active}>
235 Start Parking

237 </Layout>

241// ───────────────────── Page 5 — Active Parking ─────────────────────
242function Active({ users, session, active, setActive, history, setHistory, setSession }) {
243 const navigate = useNavigate();
244 const [now, setNow] = useState(Date.now());
245 // ✅ Real-time clock — updates every second
246 useEffect(() => {
247 const t = setInterval(() => setNow(Date.now()), 1000);
248 return () => clearInterval(t);
249 }, []);
250 
251 const startedAt = new Date(active.startedAt).getTime();
252 const elapsedMs = now - startedAt;
253 const minutes = Math.max(1, Math.ceil(elapsedMs / 60000));
254 const cost = Math.ceil((elapsedMs / 3600000) * active.pricePerHour * 100) / 100;
255 
256 function pay() {
257 const finalCost = Math.max(active.pricePerHour, cost); // minimum 1 hour
258 const entry = {
259 id: Date.now(),
260 city: active.city,
261 car: active.car,
262 startedAt: active.startedAt,
263 endedAt: new Date().toISOString(),
264 minutes,
265 cost: finalCost,
266 paid: true,
267 };
268 setHistory([...history, entry]);
269 setActive(null);
270 alert(`Paid ₪${finalCost} for ${minutes}min in ${active.city}`);
271 navigate("/history");
272 }
273 
274 function close() {
275 if (!confirm("Cancel parking without payment?")) return;
276 const entry = {
277 id: Date.now(),
278 city: active.city,
279 car: active.car,
280 startedAt: active.startedAt,
281 endedAt: new Date().toISOString(),
282 minutes,
283 cost: 0,
284 paid: false,
285 };
286 setHistory([...history, entry]);
287 setActive(null);
288 navigate("/choose-parking");
289 }
290 
291 return (
292 <Layout session={session} setSession={setSession}>
293 <h1>Active Parking</h1>
294 <section className="active-card">
295 <div className="active-pulse" aria-hidden="true">●</div>
296 <h2>{active.city}</h2>
297 <p>Car: <strong>{active.car}</strong></p>
298 <p>Rate: <strong>₪{active.pricePerHour}/h</strong></p>
299 <p>Started: <strong>{new Date(active.startedAt).toLocaleString("he-IL")}</strong></p>
300 <p>Elapsed: <strong>{minutes} min</strong></p>
301 <p className="big">Cost: <strong>₪{cost.toFixed(2)}</strong></p>
302 </section>
303 <div className="actions-row">
304 {/* ✅ FIXED: Pay only enabled when there is active parking */}
305 <button className="btn-primary" onClick={pay} disabled={!active}>
306 Pay ₪{Math.max(active.pricePerHour, cost).toFixed(2)}
307 </button>
308 <button className="btn-back" onClick={close}>Cancel without paying</button>
309 </div>
310 </Layout>
311 );
312}
