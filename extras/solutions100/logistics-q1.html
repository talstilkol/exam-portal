80// ───────────────────── Validators (per spec — EXACT messages) ─────────────────────
81// "המספר צריך להיות בעל 5 ספרות בלבד.
82// במידה והוזן מספר אשר קטן או גדול מ-5 ספרות תופיע הודעת שגיאה
83// the number must be with 5 digits."
84function validateNumber(num) {
85 if (!/^\d{5}$/.test(num)) return "the number must be with 5 digits";
86 return "";
87}
88// "שם מלא. אותיות בלבד עם רווח אחד לפחות.
89// במידה ולא יכיל רק אותיות או פחות מ-4 תווים (לא כולל הרווח):
90// the name must contain minimum 4 characters."
91function validateFullName(name) {
92 if (!name || !name.trim()) return "the name must contain minimum 4 characters";
93 // letters + spaces only, at least one space
94 if (!/^[A-Za-z]+( [A-Za-z]+)+$/.test(name.trim()))
95 return "the name must contain minimum 4 characters";
96 const lettersOnly = name.replace(/\s/g, "");
97 if (lettersOnly.length < 4)
98 return "the name must contain minimum 4 characters";
99 return "";
100}
