16// ───────────────────── Persistence ─────────────────────
17const KEY = "travel-entries";
18function loadEntries() {
19 try { const r = localStorage.getItem(KEY); if (r) return JSON.parse(r); } catch {}
20 return [];
21}
22function saveEntries(arr) { try { localStorage.setItem(KEY, JSON.stringify(arr)); } catch {} }
23 
24// ───────────────────── App shell ─────────────────────
25export default function AppShell() {
26 const [entries, setEntries] = useState(loadEntries);
27 useEffect(() => { saveEntries(entries); }, [entries]);
28 return (
29 <BrowserRouter>
30 <Routes>
31 <Route path="/" element={<JournalList entries={entries} setEntries={setEntries} />} />
32 <Route path="/add" element={<EntryForm entries={entries} setEntries={setEntries} mode="add" />} />
33 <Route path="/edit/:id" element={<EntryForm entries={entries} setEntries={setEntries} mode="edit" />} />
34 <Route path="/view/:id" element={<EntryView entries={entries} setEntries={setEntries} />} />
35 <Route path="*" element={<Navigate to="/" replace />} />
36 </Routes>
37 </BrowserRouter>
38 );
39}

96// ───────────────────── Page 2 — Entry form (add/edit) ─────────────────────
97function EntryForm({ entries, setEntries, mode }) {
98 const navigate = useNavigate();
99 const id = mode === "edit" ? Number(window.location.pathname.split("/").pop()) : null;
100 const existing = mode === "edit" ? entries.find((e) => e.id === id) : null;
101 
102 const [form, setForm] = useState(
103 existing
104 ? { ...existing, tagsInput: (existing.tags || []).join(", ") }
105 : { date: new Date().toISOString().slice(0, 10), location: "", notes: "", tagsInput: "" }
106 );
107 const [errors, setErrors] = useState({});
108 
109 function setField(field, value) {
110 setForm((f) => ({ ...f, [field]: value }));
111 if (errors[field]) {
112 setErrors((e) => { const c = { ...e }; delete c[field]; return c; });
113 }
114 }
115 
116 function submit(e) {
117 e.preventDefault();
118 const errs = {};
119 // ✅ Validation: date required and valid
120 if (!form.date) errs.date = "Date is required";
121 else if (Number.isNaN(new Date(form.date).getTime())) errs.date = "Invalid date";
122 // ✅ Validation: location required
123 if (!form.location.trim()) errs.location = "Location is required";
124 if (!form.notes.trim()) errs.notes = "Notes are required";
125 
126 if (Object.keys(errs).length > 0) { setErrors(errs); return; }
127 
128 const tags = form.tagsInput
129 .split(",")
130 .map((t) => t.trim())
131 .filter(Boolean);
132 const payload = {
133 date: form.date,
134 location: form.location.trim(),
135 notes: form.notes,
136 tags,
137 };
138 if (mode === "add") {
139 setEntries([...entries, { ...payload, id: Date.now() }]);
140 } else {
141 setEntries(entries.map((e) => e.id === id ? { ...payload, id } : e));
142 }
143 navigate("/");
144 }
145 
146 return (
147 <main className="page center">
148 <h1>{mode === "add" ? "📝 New Entry" : "✏️ Edit Entry"}</h1>
149 <form onSubmit={submit} className="entry-form" noValidate>
150 <div className={`field ${errors.date ? "has-error" : ""}`}>
151 {errors.date && <p className="err-text err-red">{errors.date}</p>}
152 <label>
153 Date:
154 <input type="date" value={form.date} onChange={(e) => setField("date", e.target.value)} />
155 </label>
156 </div>
157 <div className={`field ${errors.location ? "has-error" : ""}`}>
158 {errors.location && <p className="err-text err-red">{errors.location}</p>}
159 <label>
160 Location:
161 <input type="text" placeholder="e.g. Tokyo, Japan"
162 value={form.location} onChange={(e) => setField("location", e.target.value)} />
163 </label>
164 </div>
165 <div className={`field ${errors.notes ? "has-error" : ""}`}>
166 {errors.notes && <p className="err-text err-red">{errors.notes}</p>}
167 <label>
168 Notes:
169 <textarea rows="6" placeholder="What did you see and feel today?"
170 value={form.notes} onChange={(e) => setField("notes", e.target.value)} />
171 </label>
172 </div>
173 <label>
174 Tags (comma-separated):
175 <input type="text" placeholder="food, beach, hike"
176 value={form.tagsInput} onChange={(e) => setField("tagsInput", e.target.value)} />
177 </label>
178 <div className="row">
179 <button type="submit" className="btn-primary">{mode === "add" ? "Add" : "Save"}</button>
180 <button type="button" className="btn-back" onClick={() => navigate("/")}>← Back</button>
181 </div>
182 </form>
183 </main>
184 );
185}

187// ───────────────────── Page 3 — Entry view ─────────────────────
188function EntryView({ entries, setEntries }) {
189 const navigate = useNavigate();
190 const id = Number(window.location.pathname.split("/").pop());
191 const entry = entries.find((e) => e.id === id);
192 if (!entry) return <Navigate to="/" replace />;
193 
194 function remove() {
195 if (!confirm("Delete this entry?")) return;
196 setEntries(entries.filter((e) => e.id !== entry.id));
197 navigate("/");
198 }
199 
200 return (
201 <main className="page">
202 <header className="entry-view-head">
203 <button className="btn-back" onClick={() => navigate("/")}>← Back</button>
204 <button className="btn-edit" onClick={() => navigate(`/edit/${entry.id}`)}>✏️ Edit</button>
205 <button className="btn-danger" onClick={remove}>🗑 Delete</button>
206 </header>
207 <article className="entry-full">
208 <div className="entry-date">📅 {new Date(entry.date).toLocaleDateString("he-IL", { dateStyle: "full" })}</div>
209 <h1>📍 {entry.location}</h1>
210 {entry.tags && entry.tags.length > 0 && (
211 <div className="entry-tags">
212 {entry.tags.map((t) => <span key={t} className="tag">#{t}</span>)}
213 </div>
214 )}
215 <div className="entry-notes" style={{ whiteSpace: "pre-wrap" }}>{entry.notes}</div>
216 </article>
217 </main>
