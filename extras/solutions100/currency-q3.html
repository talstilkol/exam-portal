← חזרה לשאלה המקורית

🔧 באגים שתוקנו (לא רק תועדו — תוקנו בקוד)

  • ✅ query params הם מחרוזות — amount צריך המרה עם Number()
  • ✅ בדיקה ש־from !== to (אסור להמיר מטבע לעצמו)
  • ✅ חלוקה באפס — בדיקה ש־toRate.value !== 0
  • ✅ 404 כאשר סוג מטבע לא נמצא ב־convert

✅ דרישה-מול-מימוש (טבלת בקרה)

✓שרת Express עם express.json() + cors()
✓מערך rates[] עם אובייקטים {type, value}
✓GET /rates — כל התעריפים
✓POST /rates — הוספת תעריף, type אנגלית בלבד, value חיובי
✓PUT /rates/:type — עדכון value לפי type, 404 אם לא נמצא
✓DELETE /rates/:type — מחיקה לפי type, 404 אם לא נמצא
✓GET /convert?from=X&to=Y&amount=Z — חישוב המרה

📦 הקובץ המלא — קוד מתוקן עם hover-tip לכל שורה

1// Currency Rates — Express Q3 (20 pts)
2// Currency rate API with CRUD + conversion
3 
4const express = require("express");
5const cors = require("cors");
6 
7const app = express();
8app.use(express.json());
9app.use(cors());
10 
11let rates = [
12 { type: "USD", value: 3.5 },
13 { type: "EUR", value: 3.9 },
14 { type: "GBP", value: 4.5 },
15];
16 
17// GET /rates — return all rates
18app.get("/rates", (req, res) => {
19 res.json(rates);
20});
21 
22// POST /rates — add new rate
23app.post("/rates", (req, res) => {
24 const { type, value } = req.body;
25 if (!type || !/^[A-Za-z]+$/.test(type)) {
26 return res.status(400).json({ error: "type must be English letters only" });
27 }
28 if (typeof value !== "number" || value <= 0) {
29 return res.status(400).json({ error: "value must be a positive number" });
30 }
31 const newRate = { type: type.toUpperCase(), value };
32 rates.push(newRate);
33 res.status(201).json(newRate);
34});
35 
36// PUT /rates/:type — update value by type
37app.put("/rates/:type", (req, res) => {
38 const rate = rates.find(r => r.type === req.params.type.toUpperCase());
39 if (!rate) {
40 return res.status(404).json({ error: "Rate not found" });
41 }
42 if (typeof req.body.value !== "number" || req.body.value <= 0) {
43 return res.status(400).json({ error: "value must be a positive number" });
44 }
45 rate.value = req.body.value;
46 res.json(rate);
47});
48 
49// DELETE /rates/:type — delete by type
50app.delete("/rates/:type", (req, res) => {
51 const idx = rates.findIndex(r => r.type === req.params.type.toUpperCase());
52 if (idx === -1) {
53 return res.status(404).json({ error: "Rate not found" });
54 }
55 rates.splice(idx, 1);
56 res.json({ message: "Deleted" });
57});
58 
59// GET /convert?from=X&to=Y&amount=Z
60app.get("/convert", (req, res) => {
61 const { from, to, amount } = req.query;
62 const numAmount = Number(amount);
63 if (!from || !to || isNaN(numAmount) || numAmount <= 0) {
64 return res.status(400).json({ error: "from, to, amount are required (amount > 0)" });
65 }
66 if (from.toUpperCase() === to.toUpperCase()) {
67 return res.status(400).json({ error: "from and to must be different" });
68 }
69 const fromRate = rates.find(r => r.type === from.toUpperCase());
70 const toRate = rates.find(r => r.type === to.toUpperCase());
71 if (!fromRate || !toRate) {
72 return res.status(404).json({ error: "One or both currency types not found" });
73 }
74 if (toRate.value === 0) {
75 return res.status(400).json({ error: "Cannot divide by zero (toRate is 0)" });
76 }
77 const result = (numAmount * fromRate.value) / toRate.value;
78 res.json({ from, to, amount: numAmount, result });
79});
80 
81const PORT = 3000;
82app.listen(PORT, () => {
83 console.log(`Currency server running on http://localhost:${PORT}`);
84});
