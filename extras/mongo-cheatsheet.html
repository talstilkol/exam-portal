1. MongoDB - בסיס

▼

מה זה MongoDB?

מסד נתונים NoSQL מבוסס מסמכים (documents). במקום טבלאות ושורות (כמו ב-SQL), הנתונים נשמרים כאובייקטי JSON.

SQL (MySQL)MongoDB
DatabaseDatabase
TableCollection
RowDocument
ColumnField

מבנה Document

כל document הוא אובייקט JSON (נשמר כ-BSON - Binary JSON):

{
  "_id": ObjectId("507f1f77bcf86cd799439011"),
  "name": "Tal",
  "age": 25,
  "hobbies": ["coding", "music"],
  "address": {
    "city": "Tel Aviv",
    "zip": "61000"
  }
}

MongoDB Atlas - אחסון בענן

Atlas זה שירות ענן חינמי של MongoDB. מקבלים connection string לחיבור:

mongodb+srv://user:password@cluster0.abc123.mongodb.net/myDatabase
//          ^user  ^pass     ^cluster address             ^db name
_id נוצר אוטומטית על ידי MongoDB לכל document. זה שדה ייחודי מסוג ObjectId. לא צריך ליצור אותו ידנית.

2. חיבור Mongoose

▼

התקנה וחיבור בסיסי

// התקנה: npm install mongoose

const mongoose = require('mongoose');

const MONGO_URL = 'mongodb://127.0.0.1:27017/myapp';
// או ב-Atlas:
// const MONGO_URL = 'mongodb+srv://user:pass@cluster.mongodb.net/myapp';

mongoose.connect(MONGO_URL)
  .then(() => console.log('Connected to MongoDB'))
  .catch(err => console.error('Connection failed:', err));

חיבור עם async/await (הדפוס מהמבחנים)

async function start() {
  try {
    await mongoose.connect(MONGO_URL);
    console.log('Connected to MongoDB');
    app.listen(3000, () =>
      console.log('Server running on port 3000')
    );
  } catch (err) {
    console.error('Failed to start:', err);
    process.exit(1);
  }
}
start();

אירועי חיבור

mongoose.connection.on('connected', () => console.log('DB connected'));
mongoose.connection.on('error', err => console.error('DB error:', err));
mongoose.connection.on('disconnected', () => console.log('DB disconnected'));
במבחן, תמיד שימו את app.listen() בתוך ה-callback של החיבור ל-MongoDB, כדי שהשרת לא ירוץ לפני שיש חיבור למסד.

Schema בסיסי

const userSchema = new mongoose.Schema({
  name:      { type: String, required: true, trim: true },
  age:       { type: Number, min: 0 },
  email:     { type: String, unique: true, lowercase: true },
  isActive:  { type: Boolean, default: true },

Validators - אימות נתונים

const productSchema = new mongoose.Schema({
  name: {
    type: String,
    required: [true, 'Name is required'],  // + הודעת שגיאה
    minlength: 2,
    maxlength: 100,
    trim: true
  },
  price: {
    type: Number,
    required: true,
    min: [0, 'Price cannot be negative'],
    max: 99999
  },
  category: {
    type: String,
    enum: ['electronics', 'clothing', 'food']  // ערכים מותרים בלבד
  },
  email: {
    type: String,
    match: /^[\w.-]+@[\w.-]+\.\w+$/      // regex validation
  },
  sku: {
    type: String,
    unique: true                         // אינדקס ייחודי
  }
});

Timestamps אוטומטיים

const schema = new mongoose.Schema(
  { name: String },
  { timestamps: true }  // מוסיף createdAt + updatedAt אוטומטית
);

Nested Schema / Subdocuments

const addressSchema = new mongoose.Schema({
  city:   String,
  street: String,
  zip:    String
});

const personSchema = new mongoose.Schema({
  name:      String,
  address:   addressSchema,        // subdocument יחיד
  addresses: [addressSchema]        // מערך של subdocuments
});
unique: true הוא לא validator רגיל - הוא יוצר אינדקס ב-MongoDB. השגיאה שלו היא E11000 duplicate key ולא validation error.

4. יצירת Model

▼

Model = Class שמייצג Collection

// mongoose.model(שם יחיד, schema)
// MongoDB יוצר collection בשם רבים (users)
const User = mongoose.model('User', userSchema);
//                         'User' → collection name: 'users'
//                         'Person' → 'people'
//                         'Course' → 'courses'

מוסכמות שמות

שם Modelשם Collection
'User'users
'Student'students
'Teacher'teachers
'Course'courses
שם ה-Model תמיד Uppercase יחיד באנגלית. Mongoose ממיר אותו אוטומטית ל-lowercase רבים עבור ה-collection.

5. CRUD Operations - עם Express Routes

▼

POST Create - יצירה

// דרך 1: new + save
app.post('/users', async (req, res) => {
  try {
    const user = new User(req.body);
    await user.save();
    res.status(201).json(user);
  } catch (err) {
    res.status(400).json({ error: err.message });
  }
});

// דרך 2: Model.create (קיצור)
app.post('/users', async (req, res) => {
  try {
    const user = await User.create(req.body);
    res.status(201).json(user);
  } catch (err) {
    res.status(400).json({ error: err.message });
  }
});

GET Read - קריאה

// כל המסמכים
app.get('/users', async (req, res) => {
  const users = await User.find();
  res.json(users);
});

// לפי ID
app.get('/users/:id', async (req, res) => {
  const user = await User.findById(req.params.id);
  if (!user) return res.status(404).json({ error: 'Not found' });
  res.json(user);
});

// עם פילטר
const adults = await User.find({ age: { $gte: 18 } });

// מסמך אחד לפי שדה
const user = await User.findOne({ email: 'test@test.com' });

// בחירת שדות ספציפיים
const names = await User.find().select('name email');

// מיון והגבלה
const recent = await User.find()
  .sort({ createdAt: -1 })  // -1 = יורד (חדש ראשון)
  .limit(10);               // מקסימום 10 תוצאות

PUT Update - עדכון

// findByIdAndUpdate - הדרך הנפוצה ביותר
app.put('/users/:id', async (req, res) => {
  try {
    const updated = await User.findByIdAndUpdate(
      req.params.id,            // מי לעדכן
      req.body,                 // מה לעדכן
      { new: true, runValidators: true }  // אופציות!
    );
    if (!updated) return res.status(404).json({ error: 'Not found' });
    res.json(updated);
  } catch (err) {
    res.status(400).json({ error: err.message });
  }
});

// findOneAndUpdate - לפי שדה כלשהו (לא רק _id)
const updated = await User.findOneAndUpdate(
  { id: Number(req.params.id) },  // תנאי חיפוש
  req.body,
  { new: true, runValidators: true }
);

// updateMany - עדכון מרובה
await User.updateMany(
  { active: false },                  // תנאי
  { $set: { status: 'inactive' } }  // עדכון
);

DELETE Delete - מחיקה

app.delete('/users/:id', async (req, res) => {
  try {
    const deleted = await User.findByIdAndDelete(req.params.id);
    if (!deleted) return res.status(404).json({ error: 'Not found' });
    res.status(204).end();  // No Content
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

// מחיקת מרובים
await User.deleteMany({ age: { $lt: 18 } });
{ new: true } - בלי זה, findByIdAndUpdate מחזיר את המסמך לפני העדכון! תמיד תוסיפו את זה.
{ runValidators: true } - בלי זה, validators של ה-Schema לא רצים בעדכון!
סטטוס קודים חשובים: 201 ליצירה, 200 לקריאה/עדכון, 204 למחיקה, 404 ללא נמצא, 400 לקלט לא תקין.

6. Query Operators - אופרטורים

▼

Comparison - השוואה

Operatorמשמעותדוגמה
$eqשווה{ age: { $eq: 25 } }
$neלא שווה{ status: { $ne: 'deleted' } }
$gtגדול מ{ age: { $gt: 18 } }
$gteגדול או שווה{ age: { $gte: 18 } }
$ltקטן מ{ price: { $lt: 100 } }
$lteקטן או שווה{ price: { $lte: 50 } }
$inבתוך רשימה{ color: { $in: ['red','blue'] } }
$ninלא ברשימה{ status: { $nin: ['deleted'] } }

Logical - לוגיים

// $or - אחד מהתנאים
await User.find({
  $or: [
    { age: { $lt: 18 } },
    { status: 'minor' }
  ]
});

// $and - כל התנאים (ברוב המקרים אפשר פשוט לשים כמה שדות)
await User.find({
  $and: [
    { age: { $gte: 18 } },
    { age: { $lte: 65 } }
  ]
});

// $not - הפוך תנאי
await User.find({ age: { $not: { $gt: 30 } } });

Element

// $exists - בדיקה אם שדה קיים
await User.find({ email: { $exists: true } });

Update Operators - אופרטורי עדכון

Operatorפעולהדוגמה
$setהגדר ערך{ $set: { name: 'New' } }
$unsetהסר שדה{ $unset: { temp: '' } }
$incהגדל/הקטן מספר{ $inc: { views: 1 } }
$pushהוסף לסוף מערך{ $push: { tags: 'new' } }
$pullהסר מתוך מערך{ $pull: { tags: 'old' } }
$addToSetהוסף למערך (ללא כפילויות){ $addToSet: { tags: 'unique' } }
// דוגמה: הוסף 1 לצפיות והוסף תגית
await Post.findByIdAndUpdate(id, {
  $inc: { views: 1 },
  $push: { tags: 'trending' }
});
ב-find() אפשר פשוט לכתוב { age: 25 } במקום { age: { $eq: 25 } }. זה אותו דבר.

8. טעויות נפוצות וטיפים למבחן

▼
1. שכחתם express.json()
בלי app.use(express.json()) ה-req.body יהיה undefined! 
const app = express();
app.use(express.json());  // <-- חובה! תמיד בהתחלה
2. שכחתם { new: true }
findByIdAndUpdate מחזיר את הישן. תוסיפו { new: true } לקבל את החדש. 
// BAD - מחזיר את המסמך הישן
const user = await User.findByIdAndUpdate(id, data);

// GOOD - מחזיר את המסמך המעודכן
const user = await User.findByIdAndUpdate(id, data, { new: true });
3. חסר async/await
כל פעולות Mongoose הן אסינכרוניות. בלי await תקבלו Promise ולא תוצאה. 
// BAD - users = Promise, not array!
app.get('/users', (req, res) => {
  const users = User.find();   // חסר await!
  res.json(users);            // שולח Promise, לא data
});

// GOOD
app.get('/users', async (req, res) => {
  const users = await User.find();
  res.json(users);
});
4. חסר try/catch
בלי error handling, שגיאה תקריס את השרת. 
// BAD - if save fails, server crashes
app.post('/users', async (req, res) => {
  const user = await User.create(req.body);
  res.json(user);
});

// GOOD - with error handling
app.post('/users', async (req, res) => {
  try {
    const user = await User.create(req.body);
    res.status(201).json(user);
  } catch (err) {
    res.status(400).json({ error: err.message });
  }
});
5. סטטוס 200 במקום 201 ליצירה
201 Created הוא הסטטוס הנכון ל-POST שיוצר משאב חדש.
6. שכחתם runValidators: true
ב-update, validators של ה-Schema לא רצים כברירת מחדל!
7. Connection string ללא שם מסד נתונים 
// BAD - missing database name
'mongodb+srv://user:pass@cluster.mongodb.net/'

// GOOD
'mongodb+srv://user:pass@cluster.mongodb.net/myDatabaseName'

const express = require('express');

const PORT = 3000;
const MONGO_URL = 'mongodb://127.0.0.1:27017/exam';

// ---- Schema ----
const itemSchema = new mongoose.Schema({

app.use(express.json());

// GET all
app.get('/items', async (req, res) => {
  try {
    const all = await Item.find();
    res.json(all);
  } catch (e) { res.status(500).json({ error: e.message }); }
});

// GET by id
app.get('/items/:id', async (req, res) => {
  try {
    const item = await Item.findOne({ id: Number(req.params.id) });
    if (!item) return res.status(404).json({ error: 'Not found' });
    res.json(item);
  } catch (e) { res.status(500).json({ error: e.message }); }
});

// POST create
app.post('/items', async (req, res) => {
  try {
    const item = await Item.create(req.body);
    res.status(201).json(item);
  } catch (e) {
    if (e.code === 11000) return res.status(409).json({ error: 'Duplicate' });
    res.status(400).json({ error: e.message });
  }
});

// PUT update
app.put('/items/:id', async (req, res) => {
  try {
    const updated = await Item.findOneAndUpdate(
      { id: Number(req.params.id) },
      req.body,
      { new: true, runValidators: true }
    );
    if (!updated) return res.status(404).json({ error: 'Not found' });
    res.json(updated);
  } catch (e) { res.status(400).json({ error: e.message }); }
});

// DELETE
app.delete('/items/:id', async (req, res) => {
  try {
    const deleted = await Item.findOneAndDelete({ id: Number(req.params.id) });
    if (!deleted) return res.status(404).json({ error: 'Not found' });
    res.status(204).end();
  } catch (e) { res.status(500).json({ error: e.message }); }

// ---- Boot ----
(async () => {
  await mongoose.connect(MONGO_URL);
  app.listen(PORT, () => console.log(`Server on port ${PORT}`));

Aggregation Pipeline

שרשרת שלבים שמעבדת מסמכים ומחזירה תוצאות מחושבות:

// דוגמה: ממוצע ציונים לפי קורס
const results = await Student.aggregate([
  { $match: { active: true } },
  { $group: {
    _id: "$course",
    avgGrade: { $avg: "$grade" },
    count: { $sum: 1 }
  }},
  { $sort: { avgGrade: -1 } },
  { $project: { _id: 0, course: "$_id", avgGrade: 1, count: 1 } }
]);
Stageתפקידדוגמה
$matchסינון מסמכים (כמו find){ active: true }
$groupקיבוץ + חישוב (sum, avg, count){ _id: "$field", total: { $sum: 1 } }
$sortמיון תוצאות{ avgGrade: -1 }
$projectבחירת/שינוי שדות בפלט{ _id: 0, name: 1 }
$lookupJOIN בין collectionsראו populate למטה
$unwindפריסת מערך לשורות נפרדות"$tags"
$limitהגבלת מס׳ תוצאות10

Populate — קשרים בין Collections

הפניה (ref) למסמך ב-collection אחר, populate טוען אותו אוטומטית:

// Schema עם ref
const courseSchema = new Schema({
  name: String,
  teacher: { type: Schema.Types.ObjectId, ref: 'Teacher' },
  students: [{ type: Schema.Types.ObjectId, ref: 'Student' }]
});

// שימוש ב-populate
const course = await Course
  .findById(id)
  .populate('teacher')
  .populate('students');

// populate עם סינון שדות
await Course.findById(id)
  .populate({ path: 'students', select: 'name grade' });

// nested populate
await Course.findById(id)
  .populate({ path: 'teacher', populate: { path: 'department' } });
$lookup vs populate: — populate = Mongoose convenience (multiple queries מאחורי הקלעים). $lookup = aggregation stage (JOIN אחד ב-DB, יעיל יותר). במבחן — populate פשוט יותר לכתיבה.
