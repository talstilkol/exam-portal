React App.jsx — שלד מלא למבחן

Routes, useState, add/update/delete, form, navigate

// App.jsx — תבנית React למבחן
import { useMemo, useState } from "react";
import { BrowserRouter, Routes, Route, Link, useNavigate } from "react-router-dom";
import "./App.css";

const initialItems = [
  { id: 1, name: "Demo", active: true, amount: 10 }
];

function isText(value, min = 1) {
  return value.trim().length >= min;
}

function App() {
  const [items, setItems] = useState(initialItems);
  const [currentUser, setCurrentUser] = useState(null);

  function addItem(item) {
    setItems(prev => [...prev, { ...item, id: Date.now() }]);
  }

  function updateItem(id, updates) {
    setItems(prev => prev.map(item => item.id === id ? { ...item, ...updates } : item));
  }

  function deleteItem(id) {
    setItems(prev => prev.filter(item => item.id !== id));
  }

  return (
    <BrowserRouter>
      <header className="app-header">
        <h1>SVCollege Exam App</h1>
        <nav>
          <Link to="/">Home</Link>
          <Link to="/add">Add</Link>
        </nav>
      </header>
      <Routes>
        <Route path="/" element={<Home items={items} onDelete={deleteItem} />} />
        <Route path="/add" element={<AddPage onAdd={addItem} />} />
      </Routes>
    </BrowserRouter>
  );
}

function Home({ items, onDelete }) {
  return <main className="grid">{items.map(item => <Card key={item.id} item={item} onDelete={onDelete} />)}</main>;
}

function Card({ item, onDelete }) {
  return <section className="card"><h2>{item.name}</h2><button onClick={() => onDelete(item.id)}>Delete</button></section>;
}

function AddPage({ onAdd }) {
  const navigate = useNavigate();
  const [name, setName] = useState("");
  function submit(e) {
    e.preventDefault();
    if (!isText(name, 2)) return alert("Name must contain at least 2 chars");
    onAdd({ name, active: true, amount: 0 });
    navigate("/");
  }
  return <form onSubmit={submit}><input value={name} onChange={e => setName(e.target.value)} /><button>Add</button></form>;
}

export default App;
Express/Mongo server.js — שלד CRUD

Middleware, schema, model, GET/POST/PUT/DELETE

// server.js — תבנית Express/Mongo למבחן
import express from "express";
import cors from "cors";
import mongoose from "mongoose";

const app = express();
const PORT = 5000;

app.use(cors());
app.use(express.json());

mongoose.connect("mongodb://127.0.0.1:27017/sv_exam_db")
  .then(() => console.log("Mongo connected"))
  .catch(err => console.log(err.message));

const itemSchema = new mongoose.Schema({
  name: { type: String, required: true },
  price: { type: Number, required: true },
  category: String
});

const Item = mongoose.model("Item", itemSchema);

app.get("/api/items", async (req, res) => {
  const items = await Item.find();
  res.json(items);
});

app.post("/api/items", async (req, res) => {
  const { name, price, category } = req.body;
  if (!name || typeof price !== "number") {
    return res.status(400).json({ message: "Invalid data" });
  }
  const item = await Item.create({ name, price, category });
  res.status(201).json(item);
});

app.put("/api/items/:id", async (req, res) => {
  const item = await Item.findByIdAndUpdate(req.params.id, req.body, { new: true });
  if (!item) return res.status(404).json({ message: "Not found" });
  res.json(item);
});

app.delete("/api/items/:id", async (req, res) => {
  await Item.findByIdAndDelete(req.params.id);
  res.json({ message: "Deleted" });
});

app.listen(PORT, () => console.log(`Server running on ${PORT}`));
JavaScript algorithm — פונקציה נקייה

קלט, בדיקות, לולאה, אובייקט תוצאה, return

// JS — תבנית פונקציה אלגוריתמית למבחן
function solve(input) {
  if (!Array.isArray(input)) {
    throw new Error("Input must be an array");
  }

  const result = {};

  for (const item of input) {
    if (typeof item !== "number" || Number.isNaN(item)) {
      throw new Error("All values must be numbers");
    }

    const key = String(Math.abs(item)).length;
    result[key] = (result[key] || 0) + item;
  }

  return result;
}

console.log(solve([1, 15, 30, 5, 800]));
// { 1: 6, 2: 45, 3: 800 }

function ProductsPage() {

    }
    loadProducts();
    selectRef.current?.focus();

    ? products
    : products.filter(product => product.category === category);

}

// TypeScript — תבנית type/enum/functions
export enum Genre {
  Fiction = "Fiction",
  NonFiction = "NonFiction",
  Biography = "Biography",
  ScienceFiction = "ScienceFiction",
  Fantasy = "Fantasy"
}

export type Book = {
  title: string;
  author: string;
  publishedDate: string;
  available: boolean;
  genre: Genre;
};

export function addBook(books: Book[], book: Book): Book[] {
  return [...books, book];
}

export function searchByGenre(books: Book[], genre: Genre): Book[] {
  return books.filter(book => book.genre === genre);
}

export function updateAvailability(books: Book[], title: string, available: boolean): Book[] {
  return books.map(book => book.title === title ? { ...book, available } : book);
}

🚨 בנק טעויות — לפני שמגישים

אלה הדברים שהכי מורידים נקודות.

React — טעויות שצריך לאפס
  • לשנות מערך ישירות עם push במקום setState עם spread
  • לשכוח key ב-map
  • לכתוב class במקום className
  • לשים fetch מחוץ ל-useEffect וליצור לולאה
  • לנווט בלי BrowserRouter
  • לא להמיר input number עם Number()
  • להציג עמודים בלי Route מתאים
  • לשלוח פונקציה לילד אבל לא לקרוא לה עם הערכים הנכונים
JS — טעויות שצריך לאפס
  • לא לבדוק שהקלט הוא מערך
  • להשתמש ב-sort כשאסור
  • להחזיר console.log במקום return
  • להתבלבל בין index לערך
  • לשכוח let בתוך לולאות מקוננות
  • לא לטפל במספרים שליליים כשבודקים אורך ספרות
  • להשתמש ב-includes כשאסור בתרגיל
  • לא לזרוק שגיאה כשצריך
Express — טעויות שצריך לאפס
  • לשכוח app.use(express.json())
  • לכתוב POST בלי body parser
  • לא לשלוח status 201 ביצירה
  • לשכוח return אחרי res.status(...).json
  • להריץ server בלי listen
  • לא להתקין cors כש-frontend קורא לשרת
  • לערבב req.params ו-req.body
  • לשכוח module/type imports
Mongo — טעויות שצריך לאפס
  • Frontend מנסה להתחבר ישירות ל-Mongo
  • לא להמתין ל-await
  • לא ליצור Schema/Model
  • שמות collection/model לא עקביים
  • לא לטפל ב-404 כשאין רשומה
  • לשמור מספרים כמחרוזות בלי צורך
מבחן — טעויות שצריך לאפס
  • להתחיל בעיצוב לפני הלוגיקה
  • לא לסמן סעיפים שבוצעו
  • לא לבדוק כל ולידציה בנפרד
  • להשאיר קוד לא קשור — במבחנים זה עלול להוריד ניקוד
  • לא לשמור תיקיות לפי הוראות
  • לשכוח RAR/שם מלא/תעודת זהות לפי ההנחיות
