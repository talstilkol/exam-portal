9function validateTrip(trip) {
10 if (!/^[A-Za-z]{2,100}$/.test(trip.area)) return "Area must be English letters, 2-100 chars";
11 if (!trip.description || trip.description.length > 200) return "Description must be up to 200 chars";
12 const days = Number(trip.days);
13 if (!Number.isInteger(days) || days < 1 || days > 100) return "Days must be 1-100";
14 if (trip.partners.some(p => !/^[A-Za-z]{1,15}$/.test(p))) return "Partner names must be English letters up to 15 chars";
15 return null;
16}
17
18function App() {
19 const [logged, setLogged] = useState(false);
20 const [trips, setTrips] = useState(initialTrips);
21 return <BrowserRouter><Routes>
22 <Route path="/" element={<SignIn setLogged={setLogged} />} />
23 <Route path="/home" element={<Home logged={logged} setLogged={setLogged} trips={trips} />} />
24 <Route path="/add" element={<AddUpdate mode="add" logged={logged} setLogged={setLogged} trips={trips} setTrips={setTrips} />} />
25 <Route path="/update" element={<AddUpdate mode="update" logged={logged} setLogged={setLogged} trips={trips} setTrips={setTrips} />} />
26 </Routes></BrowserRouter>;
27}

29function Shell({ logged, setLogged, children }) {
30 const navigate = useNavigate();
31 function go(path) { if (logged) navigate(path); }
32 return <main className="app"><h1>TRAVEL-SV</h1><aside><button disabled={logged} onClick={() => navigate("/")}>SignIn</button><button disabled={!logged} onClick={() => go("/home")}>Home</button><button disabled={!logged} onClick={() => go("/add")}>Add</button><button disabled={!logged} onClick={() => go("/update")}>Update</button><button disabled={!logged} onClick={() => { setLogged(false); navigate("/"); }}>Logout</button></aside><section>{children}</section></main>;
33}

35function SignIn({ setLogged }) {
36 const [form, setForm] = useState({ username: "", password: "" });
37 const navigate = useNavigate();

39 if (form.username !== "svcollege" || form.password !== "abcde") return alert("One of the entered details is invalid");
40 setLogged(true);
41 navigate("/home");
42 }
43 return <Shell logged={false} setLogged={setLogged}><input placeholder="User Name" onChange={e => setForm({ ...form, username: e.target.value })} /><input placeholder="Password" type="password" onChange={e => setForm({ ...form, password: e.target.value })} /><button onClick={login}>Login</button></Shell>;
44}

46function Home({ logged, setLogged, trips }) {
47 const [index, setIndex] = useState(Math.max(0, trips.length - 1));
48 const trip = trips[index];
49 return <Shell logged={logged} setLogged={setLogged}>{trip ? <article className="trip"><button onClick={() => setIndex((index - 1 + trips.length) % trips.length)}>◀</button><h2>{trip.area}</h2><p>{trip.description}</p><p>{trip.days} days</p><p>{trip.partners.join(", ")}</p><button onClick={() => setIndex((index + 1) % trips.length)}>▶</button></article> : <p>No trips yet</p>}</Shell>;
50}

52function AddUpdate({ mode, logged, setLogged, trips, setTrips }) {
53 const [selectedId, setSelectedId] = useState(trips[0]?.id || "");
54 const selected = trips.find(t => t.id === Number(selectedId));
55 const [form, setForm] = useState(mode === "update" && selected ? selected : { area: "", description: "", days: "", partners: [""] });
56 const navigate = useNavigate();
57 function addPartner() { setForm({ ...form, partners: [...form.partners, ""] }); }
58 function changePartner(value, i) { setForm({ ...form, partners: form.partners.map((p, idx) => idx === i ? value : p) }); }
59 function submit() {
60 const err = validateTrip(form);
61 if (err) return alert(err);
62 const fixed = { ...form, id: form.id || Date.now(), days: Number(form.days), partners: form.partners.filter(Boolean) };
63 setTrips(mode === "add" ? [...trips, fixed] : trips.map(t => t.id === fixed.id ? fixed : t));
64 navigate("/home");
65 }
66 return <Shell logged={logged} setLogged={setLogged}>{mode === "update" ? <select value={selectedId} onChange={e => { const t = trips.find(x => x.id === Number(e.target.value)); setSelectedId(e.target.value); setForm(t); }}><option value="">Area</option>{trips.map(t => <option key={t.id} value={t.id}>{t.area}</option>)}</select> : <input placeholder="Area" value={form.area} onChange={e => setForm({ ...form, area: e.target.value })} />}
67 <textarea placeholder="Description" value={form.description} onChange={e => setForm({ ...form, description: e.target.value })} />
68 <input placeholder="Days" value={form.days} onChange={e => setForm({ ...form, days: e.target.value })} />
69 {form.partners.map((p, i) => <input key={i} placeholder="Partner" value={p} onChange={e => changePartner(e.target.value, i)} />)}<button onClick={addPartner}>+</button><button onClick={submit}>{mode === "add" ? "Add" : "Update"}</button></Shell>;
70}

9function validateTrip(trip) {
10 if (!/^[A-Za-z]{2,100}$/.test(trip.area)) return "Area must be English letters, 2-100 chars";
11 if (!trip.description || trip.description.length > 200) return "Description must be up to 200 chars";
12 const days = Number(trip.days);
13 if (!Number.isInteger(days) || days < 1 || days > 100) return "Days must be 1-100";
14 if (trip.partners.some(p => !/^[A-Za-z]{1,15}$/.test(p))) return "Partner names must be English letters up to 15 chars";
15 return null;
16}

52function AddUpdate({ mode, logged, setLogged, trips, setTrips }) {
53 const [selectedId, setSelectedId] = useState(trips[0]?.id || "");
54 const selected = trips.find(t => t.id === Number(selectedId));

56 const navigate = useNavigate();
57 function addPartner() { setForm({ ...form, partners: [...form.partners, ""] }); }
58 function changePartner(value, i) { setForm({ ...form, partners: form.partners.map((p, idx) => idx === i ? value : p) }); }
59 function submit() {

⚠️ טעויות מורחבות שלא ליפול בהן

  • כפתורי הניווט לא לחיצים לפני login תקין.
  • Home מציג את המסע האחרון ומאפשר דפדוף בחצים.
  • Add ו-Update משתמשים באותו טופס אבל פעולה שונה.
  • לא לשנות מערך/אובייקט state ישירות. תמיד ליצור עותק עם spread או map/filter.
  • לא לשכוח value/onChange בכל input אם הוא controlled component.
  • checkbox צריך checked ולא value; select צריך value מתאים.
  • מספרים שמגיעים מ-input הם מחרוזות — להמיר עם Number לפני חישוב/בדיקה.
  • לא לנווט לפני שהוולידציות עברו; navigate רק בסוף פעולה תקינה.
  • ב-map חייבים key יציב לכל כרטיס/שורה.
  • לא להציג כפתורי ניווט למשתמש לא מחובר אם הדרישה אומרת לחסום.
  • לא לשכוח להחזיר JSX אחד עוטף בכל קומפוננטה.
  • לא לשכוח להתאים route/path בדיוק לדרישה: /add, /edit, /history וכו׳.
  • אם יש חיפוש חלקי — להשתמש ב-includes על lowercase כדי לא להיכשל בגלל אותיות גדולות/קטנות.
  • כשיש עריכה — לוודא שהטופס נטען מהאובייקט הנבחר ולא נשאר ריק.
  • בעיצוב: קודם פונקציונליות, אחר כך CSS; לא לאבד נקודות על לוגיקה בגלל עיצוב מוקדם.
