1import { useState } from "react";
2import { BrowserRouter, Routes, Route, useNavigate } from "react-router-dom";
3import "./style.css";
4
5class Currency {
6 constructor(type, value) {
7 this.type = type;
8 this.value = Number(value);
9 }
10 updateValue(newValue) {
11 this.value = Number(newValue);
12 }
13}
14
15const defaults = [new Currency("DOLLAR", 4), new Currency("EURO", 5), new Currency("SHEKEL", 1)];
16
17function App() {
18 const [currencies, setCurrencies] = useState(defaults);
19 const [history, setHistory] = useState([]);
20 return <BrowserRouter><Routes>
21 <Route path="/" element={<Exchange currencies={currencies} history={history} setHistory={setHistory} />} />
22 <Route path="/update" element={<Update currencies={currencies} setCurrencies={setCurrencies} />} />
23 </Routes></BrowserRouter>;
24}
25
26function Exchange({ currencies, history, setHistory }) {
27 const [form, setForm] = useState({ amount: "", from: "", to: "" });
28 const [showList, setShowList] = useState(false);
29 const navigate = useNavigate();
30 const canStart = form.amount !== "" && form.from && form.to;
31 function convert() {
32 const from = currencies.find(c => c.type === form.from);
33 const to = currencies.find(c => c.type === form.to);
34 const amount = Number(form.amount);
35 const result = amount * from.value / to.value;
36 const item = { id: Date.now(), amount, result, from: from.type, to: to.type };
37 setHistory([...history, item]);
38 alert(`${amount} ${from.type} = ${result} ${to.type}`);
39 }
40 return <main className="calculator"><h1>Exchange</h1><input type="number" step="any" placeholder="type" onChange={e => setForm({ ...form, amount: e.target.value })} />
41 <label>From:<select onChange={e => setForm({ ...form, from: e.target.value })}><option value="">type</option>{currencies.map(c => <option key={c.type}>{c.type}</option>)}</select></label>
42 <label>To:<select onChange={e => setForm({ ...form, to: e.target.value })}><option value="">type</option>{currencies.map(c => <option key={c.type}>{c.type}</option>)}</select></label>
43 <button disabled={!canStart} onClick={convert}>start</button><button onClick={() => navigate("/update")}>Update</button><button onClick={() => location.href = "https://facebook.com"}>Share on FACEBOOK</button><button onClick={() => setShowList(!showList)}>View your exchange list</button>
44 {showList && history.map((h, index) => <article key={h.id}>#{index + 1}<br/>From {h.from} To {h.to}<br/>{h.amount} = {h.result}<button onClick={() => setHistory(history.filter(x => x.id !== h.id))}>X</button></article>)}
45 </main>;
46}
47
48function Update({ currencies, setCurrencies }) {
49 const [form, setForm] = useState({ type: "", value: "" });
50 const navigate = useNavigate();
51 const disabled = !form.type || !form.value;
52 function submit() {
53 if (!/^[A-Za-z]+$/.test(form.type)) return alert("Type must be English letters only");
54 const value = Number(form.value);
55 if (!Number.isFinite(value)) return alert("Value must be number");
56 const type = form.type.toUpperCase();
57 const exists = currencies.some(c => c.type === type);
58 setCurrencies(exists ? currencies.map(c => c.type === type ? new Currency(type, value) : c) : [...currencies, new Currency(type, value)]);
59 }
60 return <main><h1>UPDATE</h1><table><tbody>{currencies.map(c => <tr key={c.type}><td>{c.type}</td><td>{c.value}</td></tr>)}</tbody></table><input placeholder="Type" onChange={e => setForm({ ...form, type: e.target.value })} /><input type="number" placeholder="New Value" onChange={e => setForm({ ...form, value: e.target.value })} /><button disabled={disabled} onClick={submit}>UPDATE</button><button onClick={() => navigate("/")}>BACK</button></main>;
61}
62
63export default App;

⚠️ כל האפשרויות לעשות טעויות — כדי שלא תטעי

  • כפתור START לא לחיץ עד שכל הערכים קיימים
  • סוג מטבע באנגלית בלבד
  • אין להסתמך על שלושת המטבעות בלבד — קוד גנרי
  • היסטוריה צריכה להמשיך להתעדכן אחרי מחיקה

⚠️ טעויות מורחבות שלא ליפול בהן

  • Start לא לחיץ עד שיש From, To וסכום.
  • עדכון מטבע קיים לא יוצר כפילות אלא מחליף ערך.
  • המרה נשמרת ברשימה ומחיקה מעדכנת את המספור.
  • לא לשנות מערך/אובייקט state ישירות. תמיד ליצור עותק עם spread או map/filter.
  • לא לשכוח value/onChange בכל input אם הוא controlled component.
  • checkbox צריך checked ולא value; select צריך value מתאים.
  • מספרים שמגיעים מ-input הם מחרוזות — להמיר עם Number לפני חישוב/בדיקה.
  • לא לנווט לפני שהוולידציות עברו; navigate רק בסוף פעולה תקינה.
  • ב-map חייבים key יציב לכל כרטיס/שורה.
  • לא להציג כפתורי ניווט למשתמש לא מחובר אם הדרישה אומרת לחסום.
  • לא לשכוח להחזיר JSX אחד עוטף בכל קומפוננטה.
  • לא לשכוח להתאים route/path בדיוק לדרישה: /add, /edit, /history וכו׳.
  • אם יש חיפוש חלקי — להשתמש ב-includes על lowercase כדי לא להיכשל בגלל אותיות גדולות/קטנות.
  • כשיש עריכה — לוודא שהטופס נטען מהאובייקט הנבחר ולא נשאר ריק.
  • בעיצוב: קודם פונקציונליות, אחר כך CSS; לא לאבד נקודות על לוגיקה בגלל עיצוב מוקדם.
