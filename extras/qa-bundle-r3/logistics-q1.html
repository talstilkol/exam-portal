1import { useState } from "react";
2import { BrowserRouter, Routes, Route, useNavigate } from "react-router-dom";

4
5class Employee {
6 constructor(no, fullName, forklift) {
7 this.no = no;
8 this.fullName = fullName;
9 this.forklift = forklift;
10 this.visits = 0;
11 this.placed = 0;
12 }
13}
14class Product {
15 constructor(no, name, needForklift) {
16 this.no = no;
17 this.name = name;
18 this.needForklift = needForklift;
19 this.inPlace = false;
20 }
21}
22
23const initialProducts = [
24 new Product("11122", "Green Box", false),
25 new Product("22554", "Green Box", false),
26 new Product("66698", "Blue Box", true),
27 new Product("78544", "Red Box", false),
28 new Product("69875", "Red Box", false)
29];

31function App() {
32 const [workers, setWorkers] = useState([]);
33 const [products, setProducts] = useState(initialProducts);
34 const [current, setCurrent] = useState(null);
35 return <BrowserRouter><Routes>
36 <Route path="/" element={<Home />} />
37 <Route path="/signup" element={<Signup workers={workers} setWorkers={setWorkers} />} />
38 <Route path="/login" element={<Login workers={workers} setWorkers={setWorkers} current={current} setCurrent={setCurrent} products={products} setProducts={setProducts} />} />

40 </Routes></BrowserRouter>;
41}

43function Home() {
44 const navigate = useNavigate();
45 return <main className="center"><h2>Logistics Management</h2><button onClick={() => navigate("/signup")}>Sign up</button><button onClick={() => navigate("/login")}>Log in</button></main>;
46}

48function Signup({ workers, setWorkers }) {
49 const [form, setForm] = useState({ no: "", fullName: "", forklift: false });
50 const [error, setError] = useState("");
51 const navigate = useNavigate();
52 function create() {
53 if (!/^\d{5}$/.test(form.no)) return setError("the number must be with 5 digits.");
54 if (!/^[A-Za-z]+ [A-Za-z]+$/.test(form.fullName) || form.fullName.replace(" ", "").length < 4) return setError("the name must contain minimum 4 characters.");
55 if (workers.some(w => w.no === form.no)) return setError("worker already exists");
56 setWorkers([...workers, new Employee(form.no, form.fullName, form.forklift)]);
57 navigate("/");
58 }
59 return <main className="center"><h1>Sign up</h1>{error && <p className="error">{error}</p>}<input placeholder="NO." onChange={e => setForm({ ...form, no: e.target.value })} /><input placeholder="FullName" onChange={e => setForm({ ...form, fullName: e.target.value })} /><label><input type="radio" checked={!form.forklift} onChange={() => setForm({ ...form, forklift: false })} /> no</label><label><input type="radio" checked={form.forklift} onChange={() => setForm({ ...form, forklift: true })} /> yes</label><button onClick={create}>Create</button></main>;
60}
61
62function Login({ workers, setWorkers, current, setCurrent, products, setProducts }) {
63 const [no, setNo] = useState("");
64 const navigate = useNavigate();
65 function login() {
66 if (no === "99999") return navigate("/manager");
67 const found = workers.find(w => w.no === no);
68 if (!found) return alert(`worker ${no} does not exist`);
69 setWorkers(workers.map(w => w.no === no ? { ...w, visits: w.visits + 1 } : w));
70 setCurrent(no);
71 }
72 const worker = workers.find(w => w.no === current);
73 function updateProduct(p) {
74 if (p.needForklift && !worker.forklift) return alert("Need forklift truck license");
75 setProducts(products.map(x => x.no === p.no ? { ...x, inPlace: true } : x));
76 setWorkers(workers.map(w => w.no === worker.no ? { ...w, placed: w.placed + 1 } : w));
77 }
78 if (!worker) return <main className="center"><h1>Log in</h1><input placeholder="NO." onChange={e => setNo(e.target.value)} /><button onClick={login}>Enter</button></main>;
79 return <main className="dashboard"><h1>Welcome {worker.fullName}</h1><aside>Full Name: {worker.fullName}<br/>NO.: {worker.no}<br/>Forklift truck license: {worker.forklift ? "yes" : "no"}</aside><section>{products.filter(p => !p.inPlace).map(p => <article key={p.no} className="product">NO. {p.no}<br/>Name: {p.name}<br/>Need forklift truck: {p.needForklift ? "yes" : "no"}<button onClick={() => updateProduct(p)}>Update</button></article>)}</section><button onClick={() => setCurrent(null)}>Log Out</button></main>;
80}
81
82function Manager({ workers }) {
83 const navigate = useNavigate();
84 return <main><h1>Manager</h1><table><tbody>{workers.filter(w => w.visits > 0).map(w => <tr key={w.no}><td>{w.no}</td><td>{w.fullName}</td><td>{w.placed}</td></tr>)}</tbody></table><button onClick={() => navigate("/")}>Log out</button></main>;
85}
86
87export default App;

⚠️ כל האפשרויות לעשות טעויות — כדי שלא תטעי

  • NO חייב 5 ספרות בדיוק
  • שם מלא אותיות בלבד עם רווח אחד לפחות
  • ברירת מחדל forklift = no
  • המוצרים הראשונים באנגלית בלבד ואין הוספת מוצרים חדשים

⚠️ טעויות מורחבות שלא ליפול בהן

  • לא לשנות מערך/אובייקט state ישירות. תמיד ליצור עותק עם spread או map/filter.
  • לא לשכוח value/onChange בכל input אם הוא controlled component.
  • checkbox צריך checked ולא value; select צריך value מתאים.
  • מספרים שמגיעים מ-input הם מחרוזות — להמיר עם Number לפני חישוב/בדיקה.
  • לא לנווט לפני שהוולידציות עברו; navigate רק בסוף פעולה תקינה.
  • ב-map חייבים key יציב לכל כרטיס/שורה.
  • לא להציג כפתורי ניווט למשתמש לא מחובר אם הדרישה אומרת לחסום.
  • לא לשכוח להחזיר JSX אחד עוטף בכל קומפוננטה.
  • לא לשכוח להתאים route/path בדיוק לדרישה: /add, /edit, /history וכו׳.
  • אם יש חיפוש חלקי — להשתמש ב-includes על lowercase כדי לא להיכשל בגלל אותיות גדולות/קטנות.
  • כשיש עריכה — לוודא שהטופס נטען מהאובייקט הנבחר ולא נשאר ריק.
  • בעיצוב: קודם פונקציונליות, אחר כך CSS; לא לאבד נקודות על לוגיקה בגלל עיצוב מוקדם.
