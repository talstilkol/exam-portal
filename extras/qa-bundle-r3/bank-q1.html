11function validateClient(form, clients, editingId = null) {
12 if (!/^\d{9}$/.test(form.id)) return "ID must be exactly 9 digits";
13 if (!editingId && clients.some(c => c.id === form.id)) return "ID already exists";
14 if (form.username.trim().length < 4) return "User name must be at least 4 chars";
15 if (form.password.length < 6) return "Password minimum 6 chars";
16 if (form.confirm !== undefined && form.password !== form.confirm) return "Passwords must match";
17 const money = Number(form.money);
18 if (!Number.isFinite(money) || money < 0 || money > 1000000) return "Money must be 0-1000000";
19 return null;
20}

22function App() {
23 const [clients, setClients] = useState(initialClients);
24 return <BrowserRouter>
25 <Routes>
26 <Route path="/" element={<Login clients={clients} />} />
27 <Route path="/register" element={<Register clients={clients} setClients={setClients} />} />
28 <Route path="/edit/:id" element={<Register clients={clients} setClients={setClients} edit />} />
29 <Route path="/client/:id" element={<Client clients={clients} setClients={setClients} />} />
30 <Route path="/admin" element={<Admin clients={clients} setClients={setClients} />} />

32 </BrowserRouter>;

35function Login({ clients }) {
36 const [form, setForm] = useState({ username: "", password: "" });
37 const navigate = useNavigate();
38 function enter() {
39 if (form.username === "ADMIN" && form.password === "ADMIN") return navigate("/admin");
40 const client = clients.find(c => c.username === form.username && c.password === form.password);
41 if (!client) return alert("Wrong user name or password");
42 navigate(`/client/${client.id}`);
43 }
44 return <main className="center"><h1>SV-BANK</h1><input placeholder="User Name" onChange={e => setForm({ ...form, username: e.target.value })} /><input placeholder="Password" type="password" onChange={e => setForm({ ...form, password: e.target.value })} /><button onClick={() => navigate("/register")}>Create new user</button><button onClick={enter}>ENTER</button></main>;
45}

47function Register({ clients, setClients, edit = false }) {
48 const { id } = useParams();
49 const old = clients.find(c => c.id === id);
50 const [form, setForm] = useState(old ? { ...old, confirm: old.password } : { id: "", username: "", password: "", confirm: "", money: "" });
51 const navigate = useNavigate();
52 function save() {
53 const err = validateClient(form, clients, edit ? id : null);
54 if (err) return alert(err);
55 const fixed = { id: form.id, username: form.username, password: form.password, money: Number(form.money), expenses: old?.expenses || [] };
56 setClients(edit ? clients.map(c => c.id === id ? fixed : c) : [...clients, fixed]);
57 navigate(`/client/${fixed.id}`);
58 }
59 return <main className="center"><h1>REGISTER</h1>{["id", "username", "password", "confirm", "money"].map(k => <input key={k} placeholder={k} value={form[k] ?? ""} type={k.includes("password") || k === "confirm" ? "password" : "text"} onChange={e => setForm({ ...form, [k]: e.target.value })} />)}<button onClick={save}>Create</button></main>;
60}

62function Client({ clients, setClients }) {
63 const { id } = useParams();
64 const client = clients.find(c => c.id === id);
65 const [showAction, setShowAction] = useState(false);
66 const [expense, setExpense] = useState({ amount: "", company: "" });
67 const navigate = useNavigate();
68 if (!client) return <p>Client not found</p>;
69 function addExpense() {
70 const amount = Number(expense.amount);
71 setClients(clients.map(c => c.id === id ? { ...c, expenses: [...c.expenses, { company: expense.company, amount }], money: c.money - amount } : c));
72 setShowAction(false);
73 setExpense({ amount: "", company: "" });
74 }
75 return <main className="center"><h1>Welcome {client.username}</h1><button onClick={() => alert(`Balance: ${client.money}`)}>Balance</button><button onClick={() => setShowAction(true)}>ACTION</button><button onClick={() => navigate(`/edit/${id}`)}>EDIT</button><button onClick={() => navigate("/")}>EXIT</button>{showAction && <section><input type="number" placeholder="amount" onChange={e => setExpense({ ...expense, amount: e.target.value })} /><input placeholder="company" onChange={e => setExpense({ ...expense, company: e.target.value })} /><button onClick={addExpense}>Save action</button></section>}</main>;
76}

78function Admin({ clients, setClients }) {
79 const [open, setOpen] = useState(null);
80 function deleteExpense(clientId, index) {
81 setClients(clients.map(c => c.id === clientId ? { ...c, expenses: c.expenses.filter((_, i) => i !== index) } : c));
82 }
83 return <main><h1>Manager</h1>{clients.map(c => <article className="admin" key={c.id}><button onClick={() => setOpen(open === c.id ? null : c.id)}>●</button> {c.id} {c.username}{open === c.id && <section>{c.expenses.map((e, i) => <p key={i}>{e.company} {e.amount} <button onClick={() => deleteExpense(c.id, i)}>X</button></p>)}<button onClick={() => setClients(clients.filter(x => x.id !== c.id))}>Cancel</button></section>}</article>)}</main>;
84}

⚠️ כל האפשרויות לעשות טעויות — כדי שלא תטעי

  • עמוד מנהל מציג פעולות וכפתור X לכל פעולה
  • לא לשכוח רספונסיביות למובייל

⚠️ טעויות מורחבות שלא ליפול בהן

  • Admin חייב להיכנס לעמוד בנקאי, יוזר רגיל לעמוד לקוח.
  • לא לשכוח לשמור expenses כמערך בתוך הלקוח.
  • בעריכת לקוח לא ליצור לקוח חדש בטעות.
  • לא לשנות מערך/אובייקט state ישירות. תמיד ליצור עותק עם spread או map/filter.
  • לא לשכוח value/onChange בכל input אם הוא controlled component.
  • checkbox צריך checked ולא value; select צריך value מתאים.
  • מספרים שמגיעים מ-input הם מחרוזות — להמיר עם Number לפני חישוב/בדיקה.
  • לא לנווט לפני שהוולידציות עברו; navigate רק בסוף פעולה תקינה.
  • ב-map חייבים key יציב לכל כרטיס/שורה.
  • לא להציג כפתורי ניווט למשתמש לא מחובר אם הדרישה אומרת לחסום.
  • לא לשכוח להחזיר JSX אחד עוטף בכל קומפוננטה.
  • לא לשכוח להתאים route/path בדיוק לדרישה: /add, /edit, /history וכו׳.
  • אם יש חיפוש חלקי — להשתמש ב-includes על lowercase כדי לא להיכשל בגלל אותיות גדולות/קטנות.
  • כשיש עריכה — לוודא שהטופס נטען מהאובייקט הנבחר ולא נשאר ריק.
  • בעיצוב: קודם פונקציונליות, אחר כך CSS; לא לאבד נקודות על לוגיקה בגלל עיצוב מוקדם.
