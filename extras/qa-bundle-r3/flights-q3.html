1import express from "express";
2import mongoose from "mongoose";
3import cors from "cors";
4
5const app = express();
6app.use(cors());
7app.use(express.json());
8
9await mongoose.connect("mongodb://127.0.0.1:27017/students_exam");
10
11const studentSchema = new mongoose.Schema({
12 idNumber: { type: String, required: true, unique: true },
13 averageGrade: { type: Number, required: true },
14 courseName: { type: String, required: true },
15 fullName: { type: String, required: true }
16}, { collection: "students" });
17
18const Student = mongoose.model("Student", studentSchema);
19
20app.get("/students", async (req, res) => {
21 res.json(await Student.find());
22});
23
24app.get("/students/high-average", async (req, res) => {
25 res.json(await Student.find({ averageGrade: { $gt: 74 } }));
26});
27
28app.post("/students", async (req, res) => {
29 const { idNumber, averageGrade, courseName, fullName } = req.body;
30 if (!idNumber || !courseName || !fullName || !Number.isFinite(Number(averageGrade))) {
31 return res.status(400).json({ message: "ERROR: invalid data" });
32 }
33 const exists = await Student.findOne({ idNumber });
34 if (exists) return res.status(409).json({ message: "ERROR: student with this ID exists" });
35 const student = await Student.create({ idNumber, averageGrade: Number(averageGrade), courseName, fullName });
36 res.status(201).json({ message: "Student accepted successfully", student });
37});
38
39app.put("/students/:idNumber", async (req, res) => {
40 const averageGrade = Number(req.body.averageGrade);
41 if (!Number.isFinite(averageGrade)) return res.status(400).json({ message: "ERROR: invalid average" });
42 const student = await Student.findOneAndUpdate({ idNumber: req.params.idNumber }, { averageGrade }, { new: true });
43 if (!student) return res.status(404).json({ message: "ERROR: student not found" });
44 res.json({ message: "Update completed successfully", student });
45});
46
47app.listen(3000, () => console.log("Students server running on http://localhost:3000"));

⚠️ כל האפשרויות לעשות טעויות — כדי שלא תטעי

  • collection חייב להיות students
  • מעל 74 = $gt: 74
  • POST לא יוצר אם קיים אותו ID
  • PUT מחזיר ERROR אם הסטודנט לא קיים

⚠️ טעויות מורחבות שלא ליפול בהן

  • לא לשכוח app.use(express.json()) לפני routes שקוראים req.body.
  • לא לשכוח cors() אם הלקוח רץ מפורט אחר.
  • לא לכתוב route בלי res.json/res.send — כל בקשה חייבת להסתיים בתגובה.
  • ב-Mongo/Mongoose להשתמש ב-async/await ולטפל בשגיאות עם try/catch.
  • לא לבלבל בין req.params, req.query ו-req.body.
  • אם יש פילטרים אופציונליים — לבנות query בהדרגה ולא להניח שכל השדות קיימים.
  • לא לשמור סיסמאות/מידע רגיש בקוד קשיח בפרויקט אמיתי; במבחן לציין אם משתמשים בדמו.
  • לא לשכוח app.listen(PORT). בלי זה השרת לא עולה.
  • ב-POST להחזיר 201 כאשר יוצרים משאב חדש.
  • לא לשלוח node_modules בהגשה אם הדרישה אומרת לא לשלוח.
