חלק 1: Express / Node.js Patterns
Express Setup Pattern — הקמת שרת בסיסי ▼

כל שרת Express מתחיל באותה תבנית. שינון הסדר הזה הוא קריטי.

const express = require('express');
const cors = require('cors');
const app = express();

// Middleware — חייב להיות לפני ה-routes!
app.use(express.json());        // מפרסר JSON מ-body
app.use(cors());                  // מאפשר cross-origin
app.use(express.static('public')); // מגיש קבצים סטטיים

const PORT = 3000;
app.listen(PORT, () => console.log(`Server on ${PORT}`));
טעות נפוצה: שוכחים app.use(express.json()) — בלי זה req.body יהיה undefined תמיד!
טיפ למבחן: תמיד כתוב את ה-middleware לפני ה-routes. סדר הקוד = סדר הביצוע.

Custom Middleware — בדיקת Auth

const authCheck = (req, res, next) => {
  const token = req.headers['authorization'];
  if (!token) {
    return res.status(401).json({ error: 'No token provided' });
  }
  next();
};
// שימוש רק על routes ספציפיים:

Error Middleware — 4 פרמטרים!

// חייב 4 פרמטרים — ככה Express מזהה שזה error handler
app.use((err, req, res, next) => {
  console.error(err.stack);
  res.status(500).json({ error: err.message });
});
טעות נפוצה: Error middleware חייב להיות אחרי כל ה-routes! אם שמים אותו לפני — הוא לא יתפוס שגיאות.
סיכום: Middleware רץ לפי סדר הכתיבה. כל אחד חייב לקרוא ל-next() או לשלוח תשובה. Error middleware = 4 פרמטרים, תמיד בסוף.
Route Patterns — CRUD מלא ▼

GET — קריאת כל הפריטים

app.get('/items', (req, res) => {
  res.json(items);
});

GET by ID — קריאת פריט ספציפי

app.get('/items/:id', (req, res) => {
  const item = items.find(i => i.id === req.params.id);
  if (!item) return res.status(404).json({ error: 'Not found' });
  res.json(item);
});

POST — יצירת פריט חדש

app.post('/items', (req, res) => {
  const { name, price } = req.body;
  if (!name || !price) {
    return res.status(400).json({ error: 'Missing fields' });
  }
  const item = { id: Date.now().toString(), name, price };
  items.push(item);
  res.status(201).json(item);
});

PUT — עדכון פריט

app.put('/items/:id', (req, res) => {
  const index = items.findIndex(i => i.id === req.params.id);
  if (index === -1) {
    return res.status(404).json({ error: 'Not found' });
  }
  items[index] = { ...items[index], ...req.body };
  res.json(items[index]);
});

DELETE — מחיקת פריט

app.delete('/items/:id', (req, res) => {
  const index = items.findIndex(i => i.id === req.params.id);
  if (index === -1) {
    return res.status(404).json({ error: 'Not found' });
  }
  const deleted = items.splice(index, 1);
  res.json(deleted[0]);
});
טיפ למבחן: POST מחזיר 201, GET/PUT/DELETE מחזירים 200. תמיד בדוק אם הפריט קיים לפני update/delete!
טעות נפוצה: req.params.id הוא תמיד string. אם ה-id במערך הוא number, צריך להמיר: Number(req.params.id).
Filter / Search Pattern — סינון לפי תנאים (שאלה 3) ▼

דפוס חוזר במבחנים: לסנן מערך לפי מספר תנאים שמגיעים מה-body.

app.post('/items/filter', (req, res) => {
  const { yearMin, name, priceMax } = req.body;
  let result = [...items]; // העתק! לא לשנות את המקורי

  if (yearMin) {
    result = result.filter(i => i.year >= yearMin);
  }
  if (name) {
    result = result.filter(i =>
      i.name.toLowerCase().includes(name.toLowerCase())
    );
  }
  if (priceMax) {
    result = result.filter(i => i.price <= priceMax);
  }

  res.json(result);
});
טעות נפוצה: הסינון הוא AND — כל התנאים חייבים להתקיים יחד! כל if מצמצם את התוצאה הקודמת. זה לא OR.
טיפ למבחן: תמיד צור העתק עם [...items] כדי לא לפגוע במערך המקורי. השוואת שמות — תמיד עם .toLowerCase().

חלק 2: JavaScript Algorithm Patterns — שאלות Q2
Pattern 1: Accumulator / Group By — צבירה וקיבוץ ▼
שימושים: football-q2 סכום לפי אורך ספרות willing-q2 ספירת זוגיים/אי-זוגיים 
function groupByLength(arr) {
  const result = {};
  for (const num of arr) {
    const key = String(Math.abs(num)).length; // Math.abs למספרים שליליים!
    result[key] = (result[key] || 0) + num;
  }
  return result;
}

// דוגמה: groupByLength([5, 12, 300, -7, 45])
// ➜ { 1: -2, 2: 57, 3: 300 }
// key=1 → 5+(-7)=-2, key=2 → 12+45=57, key=3 → 300
טעות נפוצה: לשכוח Math.abs() עבור מספרים שליליים. בלי זה, String(-7).length = 2 (כולל המינוס) במקום 1.
טיפ למבחן: הדפוס result[key] = (result[key] || 0) + value הוא הבסיס של כל Group By. שינו רק את ה-key ואת הפעולה.
Pattern 2: Contiguous Subsequence — תת-רצף רציף ▼
שימושים: parking-q2 ספירת תת-רצפים עם סכום מטרה 
function countSubsequences(arr, target) {
  let count = 0;
  for (let i = 0; i < arr.length; i++) {
    let sum = 0;
    for (let j = i; j < arr.length; j++) {
      sum += arr[j];
      if (sum === target) count++;
    }
  }
  return count;
}

// דוגמה: countSubsequences([1, 2, 3, 0, 3], 3)
// ➜ 4
// [3], [3], [1,2], [3,0]
טיפ למבחן: שימו לב — j מתחיל מ-i (לא מ-0). זה מה שמבטיח רצף רציף. לולאה חיצונית = נקודת התחלה. לולאה פנימית = הרחבת הרצף.
Pattern 3: Matrix Traversal — סריקת מטריצה ▼
שימושים: flights-q2 ספירה במטריצה 
function countInMatrix(matrix, numbers) {
  let count = 0;
  for (let i = 0; i < matrix.length; i++) {
    for (let j = 0; j < matrix[i].length; j++) {
      if (numbers.includes(matrix[i][j])) count++;
    }
  }
  return count;
}

// דוגמה:
// countInMatrix([[1,2,3],[4,5,6],[7,8,9]], [2,5,9])
// ➜ 3
טיפ למבחן: matrix.length = מספר שורות. matrix[i].length = מספר עמודות בשורה i. אל תניח שכל השורות באותו אורך!
Pattern 4: Series Detection — זיהוי סדרה חשבונית/הנדסית ▼
שימושים: helpme-q2 זיהוי סדרה חשבונית או הנדסית 
function detectSeries(arr) {
  if (arr.length < 2) return 'none';

  const diff = arr[1] - arr[0];  // הפרש קבוע (חשבונית)
  const ratio = arr[1] / arr[0]; // מנה קבועה (הנדסית)

  let isArith = true, isGeo = true;

  for (let i = 2; i < arr.length; i++) {
    if (arr[i] - arr[i-1] !== diff) isArith = false;
    if (arr[i] / arr[i-1] !== ratio) isGeo = false;
  }

  if (isArith) return 'arithmetic';
  if (isGeo) return 'geometric';
  return 'none';
}

// דוגמה: detectSeries([2, 4, 6, 8])   ➜ 'arithmetic' (d=2)
// דוגמה: detectSeries([3, 9, 27, 81]) ➜ 'geometric' (q=3)
// דוגמה: detectSeries([1, 3, 7, 10])  ➜ 'none'
טעות נפוצה: חלוקה באפס! אם arr[0] === 0, אז ratio יהיה Infinity. צריך לטפל ב-edge case הזה.
טיפ למבחן: חשבונית = הפרש קבוע (חיבור/חיסור). הנדסית = מנה קבועה (כפל/חילוק).
Pattern 5: Digit Manipulation — מניפולציית ספרות ▼
שימושים: bank-q2 מיון ספרות logistics-q2 בינארי לעשרוני

מיון ספרות בלי .sort()

function sortDigits(num) {
  const digits = String(Math.abs(num)).split('').map(Number);
  // Bubble sort ידני
  for (let i = 0; i < digits.length; i++) {
    for (let j = i + 1; j < digits.length; j++) {
      if (digits[j] < digits[i]) {
        [digits[i], digits[j]] = [digits[j], digits[i]]; // swap
      }
    }
  }
  return Number(digits.join(''));
}

// דוגמה: sortDigits(5831) ➜ 1358
// דוגמה: sortDigits(-472) ➜ 247

בינארי לעשרוני (בלי parseInt)

function binaryToDecimal(binary) {
  let decimal = 0;
  for (let i = 0; i < binary.length; i++) {
    decimal = decimal * 2 + Number(binary[i]);
  }
  return decimal;
}

// דוגמה: binaryToDecimal("1101") ➜ 13
// 1*8 + 1*4 + 0*2 + 1*1 = 13

בדיקת פלינדרום

function isPalindrome(str) {
  return str === str.split('').reverse().join('');
}

// דוגמה: isPalindrome("racecar") ➜ true
// דוגמה: isPalindrome("hello")   ➜ false
טיפ למבחן: מיון ספרות: String → split → map(Number) → sort → join → Number. בינארי: decimal = decimal * 2 + currentBit.
Pattern 6: Subsequence Check — בדיקת תת-סדרה ▼
שימושים: travel-q2 בדיקה אם מערך מופיע כתת-סדרה בסדר 
function isSubsequence(arr, subArr) {
  let j = 0; // מצביע על subArr
  for (let i = 0; i < arr.length && j < subArr.length; i++) {
    if (arr[i] === subArr[j]) j++;
  }
  return j === subArr.length;
}

// דוגמה: isSubsequence([1,5,2,8,3,9], [5,8,9]) ➜ true
// 5 מופיע לפני 8, ו-8 לפני 9 (לא חייב רצוף!)
// דוגמה: isSubsequence([1,5,2,8,3,9], [8,5,9]) ➜ false
// 8 מופיע אחרי 5, לא לפני — הסדר לא תואם
טעות נפוצה: תת-סדרה (subsequence) ≠ תת-רצף (subarray)! תת-סדרה שומרת על סדר אבל לא חייבת להיות רציפה.
טיפ למבחן: שני מצביעים: i על המערך הגדול, j על הקטן. מקדמים את j רק כשיש התאמה. אם j הגיע לסוף — מצאנו.
Pattern 7: Modified Fibonacci — סדרת פיבונאצ'י ▼
שימושים: currency-q2 מציאת מיקום בסדרה מסוג פיבונאצ'י 
// סדרת פיבונאצ'י קלאסית
function fibonacci(n) {
  if (n <= 0) return 0;
  if (n === 1) return 1;
  let a = 0, b = 1;
  for (let i = 2; i <= n; i++) {
    [a, b] = [b, a + b];
  }
  return b;
}

// מציאת מיקום מספר בסדרת פיבונאצ'י
function findFibPosition(target) {
  if (target < 0) return -1;
  let a = 0, b = 1, pos = 0;
  while (a < target) {
    [a, b] = [b, a + b];
    pos++;
  }
  return a === target ? pos : -1;
}

// דוגמה: findFibPosition(8) ➜ 6
// סדרה: 0,1,1,2,3,5,8 — מיקום 6
// דוגמה: findFibPosition(7) ➜ -1 (לא בסדרה)
טיפ למבחן: עבודה עם שני משתנים a, b וה-swap הוא [a, b] = [b, a + b]. לא צריך רקורסיה — לולאה פשוטה ויעילה יותר.
Edge Cases & Validation — מקרי קצה וולידציה ▼ 
// 1. מערך ריק
if (!arr || arr.length === 0) return defaultValue;

// 2. מספרים שליליים — ספרות
const digits = String(Math.abs(num));

// 3. בדיקת NaN
if (typeof x !== 'number' || isNaN(x)) {
  throw new Error('Invalid input');
}

// 4. בדיקה שזה מערך
if (!Array.isArray(arr)) throw new Error('Expected array');

// 5. 0 vs -0 (edge case נדיר)
Object.is(0, -0); // false
0 === -0;          // true (זהירות!)

// 6. חלוקה באפס
if (denominator === 0) return 'Cannot divide by zero';
טעות נפוצה: לא לבדוק קלט לפני עיבוד. תמיד תתחילו עם validations! במיוחד: מערך ריק, null/undefined, סוג שגוי.
Exam Strategy — אסטרטגיית מבחן לשאלות JS ▼
  1. קראו את השאלה פעמיים — הבינו מה הקלט ומה הפלט המצופה.
  2. כתבו את חתימת הפונקציה קודם — function solve(arr, target)
  3. הוסיפו validations — מערך ריק? null? סוג שגוי?
  4. טפלו ב-edge cases — מספרים שליליים, 0, מערך באורך 1.
  5. בדקו עם הדוגמה שבשאלה — רוצו את הפונקציה בראש.
  6. בדקו עם edge case — מערך ריק, מספר שלילי, אפס.
טיפ למבחן: אם לא בטוחים באלגוריתם — תתחילו עם brute force (לולאות מקוננות). עדיף תשובה נכונה ואיטית מאשר תשובה לא נכונה ומהירה.
זיהוי Pattern לפי סוג השאלה:
"קבצו/סכמו לפי X" → Accumulator
"ספרו תת-רצפים עם סכום" → Contiguous Subsequence
"סרקו מטריצה" → Matrix Traversal
"סדרה חשבונית/הנדסית?" → Series Detection
"מיינו ספרות / המירו בסיסים" → Digit Manipulation
"האם X מופיע בסדר ב-Y?" → Subsequence Check
"מצאו מיקום בסדרה" → Modified Fibonacci
